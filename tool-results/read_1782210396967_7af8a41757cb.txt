    34→  /// Get movies with cursor-based pagination
    35→  Future<Map<String, dynamic>> getMovies({
    36→    int limit = 50,
    37→    DocumentSnapshot? startAfter,
    38→  }) async {
    39→    // === PRIMARY: orderBy('updatedAt', descending) — admin edits bump to top ===
    40→    // Requires composite index (type ASC, updatedAt DESC). Declared in
    41→    // firestore.indexes.json. If the index isn't deployed yet (e.g., Bro
    42→    // hasn't run `firebase deploy --only firestore:indexes`), this query
    43→    // throws and we fall through to the SECONDARY path below.
    44→    try {
    45→      Query query = _moviesRef
    46→          .where('type', isEqualTo: 'movie')
    47→          .orderBy('updatedAt', descending: true)
    48→          .limit(limit);
    49→
    50→      if (startAfter != null) {
    51→        query = query.startAfterDocument(startAfter);
    52→      }
    53→
    54→      final snapshot = await query.get();
    55→      final movies = snapshot.docs
    56→          .map((doc) => Movie.fromMap(
    57→                doc.data() as Map<String, dynamic>,
    58→                docId: doc.id,
    59→              ))
    60→          .toList();
    61→
    62→      return {
    63→        'movies': movies,
    64→        'hasMore': snapshot.docs.length >= limit,
    65→        'lastDoc': snapshot.docs.isNotEmpty ? snapshot.docs.last : null,
    66→      };
    67→    } catch (e) {
    68→      debugPrint('getMovies: primary orderBy(updatedAt) failed: $e — trying createdAt fallback');
    69→    }
    70→
    71→    // === SECONDARY: orderBy('createdAt', descending) ===
    72→    // Uses the EXISTING composite index (type ASC, createdAt DESC) declared
    73→    // in firestore.indexes.json. New movies (created via TMDB Generator /
    74→    // Admin Panel / Batch Import) all have createdAt set to server
    75→    // timestamp at creation time, so this puts newly-added movies at the
    76→    // top of the list — fixing Bro's 'TMDB Generator new movie doesn't
    77→    // appear at top of Home' bug.
    78→    //
    79→    // Trade-off vs primary: admin EDITS won't bump a movie to the top
    80→    // (createdAt is preserved on update, only updatedAt changes). That's
    81→    // an acceptable degradation when the updatedAt index isn't deployed
    82→    // yet — better than showing a stale random sample.
    83→    try {
    84→      Query query = _moviesRef
    85→          .where('type', isEqualTo: 'movie')
    86→          .orderBy('createdAt', descending: true)
    87→          .limit(limit);
    88→
    89→      if (startAfter != null) {
    90→        query = query.startAfterDocument(startAfter);
    91→      }
    92→
    93→      final snapshot = await query.get();
    94→      final movies = snapshot.docs
    95→          .map((doc) => Movie.fromMap(
    96→                doc.data() as Map<String, dynamic>,
    97→                docId: doc.id,
    98→              ))
    99→          .toList();
   100→
   101→      return {
   102→        'movies': movies,
   103→        'hasMore': snapshot.docs.length >= limit,
   104→        'lastDoc': snapshot.docs.isNotEmpty ? snapshot.docs.last : null,
   105→      };
   106→    } catch (e) {
   107→      debugPrint('getMovies: secondary orderBy(createdAt) also failed: $e — falling back to no-orderBy');
   108→    }
   109→
   110→    // === TERTIARY: no orderBy — last-resort for legacy docs missing both
   111→    //     timestamps. Fetches by document ID order (essentially arbitrary)
   112→    //     and sorts client-side. With small `limit` (e.g., Home's 10),
   113→    //     this returns a non-representative sample — new movies likely
   114→    //     missing. Acceptable only when primary and secondary both fail.
   115→    try {
   116→      Query query = _moviesRef
   117→          .where('type', isEqualTo: 'movie')
   118→          .limit(limit);
   119→
   120→      if (startAfter != null) {
   121→        query = query.startAfterDocument(startAfter);
   122→      }
   123→
   124→      final snapshot = await query.get();
   125→      final movies = snapshot.docs
   126→          .map((doc) => Movie.fromMap(
   127→                doc.data() as Map<String, dynamic>,
   128→                docId: doc.id,
   129→              ))
   130→          .toList();
   131→
   132→      // Sort client-side by updatedAt (falls back to createdAt if updatedAt missing)
   133→      movies.sort((a, b) {
   134→        final aDate = b.updatedAt ?? b.createdAt ?? DateTime(2000);
   135→        final bDate = a.updatedAt ?? a.createdAt ?? DateTime(2000);
   136→        return aDate.compareTo(bDate);
   137→      });
   138→
   139→      return {
   140→        'movies': movies,
   141→        'hasMore': snapshot.docs.length >= limit,
   142→        'lastDoc': snapshot.docs.isNotEmpty ? snapshot.docs.last : null,
   143→      };
   144→    } catch (e2) {
   145→      debugPrint('getMovies: no-orderBy fallback also failed: $e2');
   146→      return {
   147→        'movies': <Movie>[],
   148→        'hasMore': false,
   149→        'lastDoc': null,
   150→      };
   151→    }
   152→  }

   260→  /// Get all movies and series (for admin panel).
   261→  ///
   262→  /// Robustness strategy mirrors getMovies() / getSeries():
   263→  ///   1. PRIMARY  — orderBy('updatedAt', descending). This is the same field
   264→  ///      Home uses, so movies that appear on Home will appear here too.
   265→  ///   2. FALLBACK — If primary throws (e.g., missing index) OR returns 0
   266→  ///      docs (e.g., legacy movies that have neither 'updatedAt' nor
   267→  ///      'createdAt' because they were created via an older code path),
   268→  ///      retry without orderBy and sort client-side. This guarantees that
   269→  ///      the Admin Panel NEVER shows an empty list when movies exist.
   270→  ///
   271→  /// Background: a previous regression caused the Admin Panel → All tab to
   272→  /// appear empty even though Home showed 1068 movies. Root cause was that
   273→  /// this method used orderBy('createdAt') with no fallback, so any movie
   274→  /// missing the 'createdAt' field was silently excluded from the query
   275→  /// result. Search still worked because searchAllPosts() orders by
   276→  /// 'title_lowercase' (a different field that those movies did have).
   277→  /// This fix aligns getAllPosts() with getMovies() so the three code paths
   278→  /// no longer disagree on which movies "exist".
   279→  Future<Map<String, dynamic>> getAllPosts({
   280→    int limit = 50,
   281→    DocumentSnapshot? startAfter,
   282→  }) async {
   283→    // === PRIMARY: orderBy('updatedAt') — same field Home uses ===
   284→    try {
   285→      Query query = _moviesRef
   286→          .orderBy('updatedAt', descending: true)
   287→          .limit(limit);
   288→
   289→      if (startAfter != null) {
   290→        query = query.startAfterDocument(startAfter);
   291→      }
   292→
   293→      final snapshot = await query.get();
   294→      final movies = snapshot.docs
   295→          .map((doc) => Movie.fromMap(
   296→                doc.data() as Map<String, dynamic>,
   297→                docId: doc.id,
   298→              ))
   299→          .toList();
   300→
   301→      // If we got at least one movie, or we got fewer than `limit` (meaning
   302→      // we've reached the end of the collection), this is a trustworthy
   303→      // result — return it.
   304→      if (movies.isNotEmpty || snapshot.docs.length < limit) {
   305→        return {
   306→          'movies': movies,
   307→          'hasMore': snapshot.docs.length >= limit,
   308→          'lastDoc': snapshot.docs.isNotEmpty ? snapshot.docs.last : null,
   309→        };
   310→      }
   311→
   312→      // movies is empty AND snapshot.docs.length >= limit — suspicious.
   313→      // This shouldn't normally happen, but if it does (e.g., all docs in
   314→      // this page were skipped by Movie.fromMap throwing — unlikely), fall
   315→      // through to the no-orderBy fallback below so we don't return empty.
   316→      debugPrint('getAllPosts: primary returned 0 movies but limit=$limit — falling through to no-orderBy fallback');
   317→    } catch (e) {
   318→      // Most likely cause: missing composite index (shouldn't happen here
   319→      // since we have no `where` clause, but the catch is cheap insurance).
   320→      debugPrint('getAllPosts: primary orderBy(updatedAt) failed: $e — falling back to no-orderBy query');
   321→    }
   322→
   323→    // === FALLBACK: no orderBy — fetch by document ID order, sort client-side ===
   324→    // This handles the case where movies exist but lack 'updatedAt' (e.g.,
   325→    // legacy imports). Without orderBy, Firestore returns docs in document
   326→    // ID order, which is stable enough for cursor-based pagination to work.
   327→    try {
   328→      Query query = _moviesRef.limit(limit);
   329→
   330→      if (startAfter != null) {
   331→        query = query.startAfterDocument(startAfter);
   332→      }
   333→
   334→      final snapshot = await query.get();
   335→      final movies = snapshot.docs
   336→          .map((doc) => Movie.fromMap(
   337→                doc.data() as Map<String, dynamic>,
   338→                docId: doc.id,
   339→              ))
   340→          .toList();
   341→
   342→      // Sort client-side by updatedAt (falls back to createdAt if missing).
   343→      // Movies with neither field sort to the bottom (DateTime(2000)).
   344→      movies.sort((a, b) {
   345→        final aDate = b.updatedAt ?? b.createdAt ?? DateTime(2000);
   346→        final bDate = a.updatedAt ?? a.createdAt ?? DateTime(2000);
   347→        return aDate.compareTo(bDate);
   348→      });
   349→
   350→      return {
   351→        'movies': movies,
   352→        'hasMore': snapshot.docs.length >= limit,
   353→        'lastDoc': snapshot.docs.isNotEmpty ? snapshot.docs.last : null,
   354→      };
   355→    } catch (e2) {
   356→      debugPrint('getAllPosts: no-orderBy fallback also failed: $e2');
   357→      return {
   358→        'movies': <Movie>[],
   359→        'hasMore': false,
   360→        'lastDoc': null,
   361→      };
   362→    }
   363→  }

   365→  /// Get the REAL total counts of movies/series in Firestore (for Admin Panel
   366→  /// tab labels). Uses Firestore's AggregateQuery.count() which is billed as
   367→  /// a single document read regardless of collection size — cheap and fast.
   368→  ///
   369→  /// Returns a map with keys:
   370→  ///   - 'all'    : total count of documents in the movies collection
   371→  ///   - 'movies' : count where type == 'movie'
   372→  ///   - 'series' : count where type == 'series'
   373→  ///
   374→  /// If any of the count queries fail (e.g., permission denied), the value
   375→  /// is returned as 0 — the caller can fall back to the page-loaded count.
   376→  Future<Map<String, int>> getTotalPostCounts() async {
   377→    try {
   378→      final allCount = await _moviesRef.count().get();
   379→      final moviesCount = await _moviesRef
   380→          .where('type', isEqualTo: 'movie')
   381→          .count()
   382→          .get();
   383→      final seriesCount = await _moviesRef
   384→          .where('type', isEqualTo: 'series')
   385→          .count()
   386→          .get();
   387→      return {
   388→        'all': allCount.count ?? 0,
   389→        'movies': moviesCount.count ?? 0,
   390→        'series': seriesCount.count ?? 0,
   391→      };
   392→    } catch (e) {
   393→      debugPrint('getTotalPostCounts failed: $e');
   394→      return {'all': 0, 'movies': 0, 'series': 0};
   395→    }
   396→  }

   398→  /// Search all posts (for admin panel)
   399→  /// Uses 'title_lowercase' field for case-insensitive prefix search
   400→  /// with substring fallback for short queries (1-2 chars) so that
   401→  /// single-character searches like "o" still return matches.
   402→  Future<List<Movie>> searchAllPosts(String keyword) async {
   403→    if (keyword.trim().isEmpty) return [];
   404→
   405→    final lowerKeyword = keyword.toLowerCase().trim();
   406→    final upperKeyword = lowerKeyword + '\uf8ff';
   407→
   408→    try {
   409→      // Primary: search on 'title_lowercase' field (case-insensitive prefix match)
   410→      final snapshot = await _moviesRef
   411→          .where('title_lowercase', isGreaterThanOrEqualTo: lowerKeyword)
   412→          .where('title_lowercase', isLessThanOrEqualTo: upperKeyword)
   413→          .orderBy('title_lowercase')
   414→          .limit(50)
   415→          .get();
   416→
   417→      final prefixResults = snapshot.docs
   418→          .map((doc) => Movie.fromMap(
   419→                doc.data() as Map<String, dynamic>,
   420→                docId: doc.id,
   421→              ))
   422→          .toList();
   423→
   424→      // SUBSTRING FALLBACK for short queries (<=2 chars).
   425→      // Same rationale as _searchWithKeyword Strategy 1.5 — see comment there.
   426→      //
   427→      // Always fires for short queries (length <= 2) regardless of whether
   428→      // the prefix search returned matches. This catches titles like
   429→      // "Spider-Man" or "Thor" when the user types "o" — these don't START
   430→      // with "o" so prefix search misses them, but they CONTAIN "o" so the
   431→      // substring fallback finds them. The previous version only fired this
   432→      // fallback when prefixResults was empty, which meant single-letter
   433→      // searches like "o" only returned movies starting with "o" (Ocean's,
   434→      // Once Upon a Time, etc.) — Bro reported this exact bug on the Admin
   435→      // Panel Tab too. Task 25.
   436→      if (lowerKeyword.length <= 2) {
   437→        try {
   438→          final subSnapshot = await _moviesRef.limit(200).get();
   439→          final subResults = subSnapshot.docs
   440→              .map((doc) => Movie.fromMap(
   441→                    doc.data() as Map<String, dynamic>,
   442→                    docId: doc.id,
   443→                  ))
   444→              .where((m) => m.titleLowercase.contains(lowerKeyword))
   445→              .toList();
   446→          // Merge prefix + substring, dedup by ID.
   447→          final seenIds = <String>{};
   448→          final merged = <Movie>[];
   449→          for (final m in [...prefixResults, ...subResults]) {
   450→            if (!seenIds.contains(m.id)) {
   451→              seenIds.add(m.id);
   452→              merged.add(m);
   453→            }
   454→          }
   455→          return merged;
   456→        } catch (_) {
   457→          return prefixResults;
   458→        }
   459→      }
   460→
   461→      return prefixResults;
   462→    } catch (e) {
   463→      // Fallback: if title_lowercase index doesn't exist, try old 'title' field
   464→      debugPrint('searchAllPosts with title_lowercase failed, trying fallback: $e');
   465→      try {
   466→        final snapshot = await _moviesRef
   467→            .where('title', isGreaterThanOrEqualTo: lowerKeyword)
   468→            .where('title', isLessThanOrEqualTo: upperKeyword)
   469→            .orderBy('title')
   470→            .limit(50)
   471→            .get();
   472→
   473→        return snapshot.docs
   474→            .map((doc) => Movie.fromMap(
   475→                  doc.data() as Map<String, dynamic>,
   476→                  docId: doc.id,
   477→                ))
   478→            .toList();
   479→      } catch (e2) {
   480→        debugPrint('searchAllPosts fallback also failed: $e2');
   481→        return [];
   482→      }
   483→    }
   484→  }

   652→  /// Batch-fetch latest Movie data for a list of movie IDs.
   653→  ///
   654→  /// Used by Bookmark/Recent pages to refresh stale cached ratings:
   655→  /// when a user bookmarks a movie, the Movie object is snapshotted to
   656→  /// Firestore/local storage at that moment. If the admin later edits
   657→  /// the rating (or any field), the cached copy stays stale. This method
   658→  /// re-fetches the latest Movie data in a single Firestore call (using
   659→  /// 'in' query — up to 30 IDs per call, Firestore limit) so the UI can
   660→  /// display the current rating instead of the cached "N/A".
   661→  ///
   662→  /// Returns a Map<movieId, Movie> for O(1) lookup. Movies that no
   663→  /// longer exist are silently skipped (their IDs won't appear in the
   664→  /// returned map).
   665→  ///
   666→  /// Cost: 1 read per movie ID, batched into 1 query per 30 IDs.
   667→  /// For typical bookmark/recent lists (10-50 items), this is 1-2 queries.
   668→  Future<Map<String, Movie>> getMoviesByIds(List<String> ids) async {
   669→    if (ids.isEmpty) return {};
   670→    final result = <String, Movie>{};
   671→    // Firestore 'in' query supports max 30 values per call.
   672→    final chunks = <List<String>>[];
   673→    for (var i = 0; i < ids.length; i += 30) {
   674→      chunks.add(ids.sublist(i, (i + 30).clamp(0, ids.length)));
   675→    }
   676→    try {
   677→      final futures = chunks.map((chunk) async {
   678→        final snap = await _moviesRef.where(FieldPath.documentId, whereIn: chunk).get();
   679→        for (final doc in snap.docs) {
   680→          final movie = Movie.fromMap(doc.data() as Map<String, dynamic>, docId: doc.id);
   681→          result[movie.id] = movie;
   682→        }
   683→      });
   684→      await Future.wait(futures);
   685→    } catch (e) {
   686→      debugPrint('getMoviesByIds failed: $e');
   687→    }
   688→    return result;
   689→  }

   845→  /// Search movies with multiple filters (client-side for reliability)
   846→  /// Supports: keyword, genre, type, year, rating, and sorting
   847→  /// Advanced search: supports Name + Year combined search (e.g., "Kung Fu Panda 2008")
   848→  /// by splitting query into tokens and matching all tokens against title/year
   849→  Future<Map<String, dynamic>> searchMoviesWithFilters({
   850→    String? keyword,
   851→    String? genre,
   852→    String? type, // 'movie' or 'series'
   853→    String? year,
   854→    String? rating, // e.g. '7', '8' - minimum rating
   855→    String? sortBy, // 'latest', 'rating', 'name'
   856→    int limit = 50,
   857→    DocumentSnapshot? startAfter,
   858→  }) async {
   859→    try {
   860→      // Determine if we need keyword-based search (more extensive fetching)
   861→      final hasKeyword = keyword != null && keyword.trim().isNotEmpty;
   862→
   863→      if (hasKeyword) {
   864→        // For keyword searches, we need to fetch more documents to find matches
   865→        // especially for old movies that may not appear in the first batch
   866→        return _searchWithKeyword(
   867→          keyword: keyword!,
   868→          genre: genre,
   869→          type: type,
   870→          year: year,
   871→          rating: rating,
   872→          sortBy: sortBy,
   873→          limit: limit,
   874→          startAfter: startAfter,
   875→        );
   876→      }
   877→
   878→      // No keyword — just filter-based browsing with pagination
   879→      Query query = _moviesRef;
   880→
   881→      // Apply server-side filters where possible
   882→      if (type != null && type.isNotEmpty) {
   883→        query = query.where('type', isEqualTo: type);
   884→      }
   885→      if (genre != null && genre.isNotEmpty) {
   886→        query = query.where('categories', arrayContains: genre);
   887→      }
   888→
   889→      // Order by - only if we don't have genre filter (composite index issue)
   890→      if (sortBy == 'rating') {
   891→        query = query.orderBy('rating', descending: true);
   892→      } else if (sortBy == 'name') {
   893→        query = query.orderBy('title');
   894→      } else {
   895→        if (genre == null || genre.isEmpty) {
   896→          query = query.orderBy('createdAt', descending: true);
   897→        }
   898→      }
   899→
   900→      // Apply cursor-based pagination
   901→      if (startAfter != null) {
   902→        query = query.startAfterDocument(startAfter);
   903→      }
   904→
   905→      query = query.limit(limit);
   906→
   907→      final snapshot = await query.get();
   908→      var allMovies = snapshot.docs
   909→          .map((doc) => Movie.fromMap(
   910→                doc.data() as Map<String, dynamic>,
   911→                docId: doc.id,
   912→              ))
   913→          .toList();
   914→
   915→      // Client-side filtering for fields that can't be queried together in Firestore
   916→      var filtered = allMovies;
   917→
   918→      // Type filter (client-side fallback if not applied server-side)
   919→      if (type != null && type.isNotEmpty && genre != null && genre.isNotEmpty) {
   920→        filtered = filtered.where((m) => m.type == type).toList();
   921→      }
   922→
   923→      // Year filter
   924→      if (year != null && year.isNotEmpty) {
   925→        filtered = filtered.where((m) => m.year == year).toList();
   926→      }
   927→
   928→      // Rating filter (minimum rating)
   929→      if (rating != null && rating.isNotEmpty) {
   930→        final minRating = double.tryParse(rating) ?? 0.0;
   931→        filtered = filtered.where((m) {
   932→          final movieRating = double.tryParse(m.rating ?? '0') ?? 0.0;
   933→          return movieRating >= minRating;
   934→        }).toList();
   935→      }
   936→
   937→      // Client-side sort fallback for genre queries
   938→      if (genre != null && genre.isNotEmpty) {
   939→        if (sortBy == 'rating') {
   940→          filtered.sort((a, b) =>
   941→              (double.tryParse(b.rating ?? '0') ?? 0.0)
   942→              .compareTo(double.tryParse(a.rating ?? '0') ?? 0.0));
   943→        } else if (sortBy == 'name') {
   944→          filtered.sort((a, b) => a.title.toLowerCase().compareTo(b.title.toLowerCase()));
   945→        } else {
   946→          filtered.sort((a, b) => (b.createdAt ?? DateTime(2000))
   947→              .compareTo(a.createdAt ?? DateTime(2000)));
   948→        }
   949→      }
   950→
   951→      return {
   952→        'movies': filtered.take(limit).toList(),
   953→        'hasMore': allMovies.length >= limit,
   954→        'lastDoc': snapshot.docs.isNotEmpty ? snapshot.docs.last : null,
   955→      };
   956→    } catch (e) {
   957→      debugPrint('searchMoviesWithFilters failed: $e');
   958→      return {
   959→        'movies': <Movie>[],
   960→        'hasMore': false,
   961→        'lastDoc': null,
   962→      };
   963→    }
   964→  }
   965→
   966→  /// Internal: keyword-based search that fetches more documents to find old movies too
   967→  /// Supports advanced token-based search: "Kung Fu Panda 2008" matches title AND year
   968→  Future<Map<String, dynamic>> _searchWithKeyword({
   969→    required String keyword,
   970→    String? genre,
   971→    String? type,
   972→    String? year,
   973→    String? rating,
   974→    String? sortBy,
   975→    int limit = 50,
   976→    DocumentSnapshot? startAfter,
   977→  }) async {
   978→    // Split keyword into tokens for advanced search
   979→    final rawTokens = keyword.toLowerCase().trim().split(RegExp(r'\s+'));
   980→    // Separate year-like tokens (4 digits) from name tokens
   981→    final yearTokens = <String>[];
   982→    final nameTokens = <String>[];
   983→    for (final token in rawTokens) {
   984→      if (RegExp(r'^\d{4}$').hasMatch(token)) {
   985→        yearTokens.add(token);
   986→      } else {
   987→        nameTokens.add(token);
   988→      }
   989→    }
   990→
   991→    // =========================================================================
   992→    // OPTIMIZATION (mirrors the legacy searchMovies() pattern, now removed)
   993→    // =========================================================================
   994→    // Before: 2 + 2N queries/search (N = name token count) = up to ~500 reads
   995→    //   prefix + (per-token search_keywords) + orderBy(updatedAt) + no-orderBy
   996→    // After:  1 + N queries/search = ~100-200 reads (50-75% reduction)
   997→    //   Broader fallbacks are skipped when the primary queries returned
   998→    //   ANY matching result. Only legacy movies missing both
   999→    //   'title_lowercase' and 'search_keywords' trigger the fallback.
  1000→    // =========================================================================
  1001→
  1002→    // Build a comprehensive search query
  1003→    // Fetch a large batch to ensure old movies are included
  1004→    final fetchLimit = (limit * 3).clamp(60, 200);
  1005→
  1006→    try {
  1007→      // Strategy 1: Firestore prefix search on 'title_lowercase' (case-insensitive)
  1008→      final lowerKeyword = keyword.toLowerCase().trim();
  1009→      final upperKeyword = lowerKeyword + '\uf8ff';
  1010→      final prefixSnapshot = await _moviesRef
  1011→          .where('title_lowercase', isGreaterThanOrEqualTo: lowerKeyword)
  1012→          .where('title_lowercase', isLessThanOrEqualTo: upperKeyword)
  1013→          .orderBy('title_lowercase')
  1014→          .limit(fetchLimit)
  1015→          .get();
  1016→
  1017→      final prefixMovies = prefixSnapshot.docs
  1018→          .map((doc) => Movie.fromMap(
  1019→                doc.data() as Map<String, dynamic>,
  1020→                docId: doc.id,
  1021→              ))
  1022→          .toList();
  1023→
  1024→      // Strategy 2: search_keywords array for each name token (non-prefix word matching)
  1025→      // e.g. searching "Avengers" finds "The Avengers" because "avengers" is a search_keyword
  1026→      final keywordResults = <Movie>[];
  1027→      final keywordSeenIds = <String>{};
  1028→      for (final token in nameTokens) {
  1029→        try {
  1030→          final kwSnapshot = await _moviesRef
  1031→              .where('search_keywords', arrayContains: token)
  1032→              .limit(fetchLimit)
  1033→              .get();
  1034→          for (final doc in kwSnapshot.docs) {
  1035→            final movie = Movie.fromMap(doc.data() as Map<String, dynamic>, docId: doc.id);
  1036→            if (!keywordSeenIds.contains(movie.id)) {
  1037→              keywordSeenIds.add(movie.id);
  1038→              keywordResults.add(movie);
  1039→            }
  1040→          }
  1041→        } catch (_) {
  1042→          // search_keywords field may not exist yet, skip gracefully
  1043→        }
  1044→      }
  1045→
  1046→      // Strategy 1.5: SUBSTRING FALLBACK for short queries (1-2 chars)
  1047→      // =========================================================================
  1048→      // Bro reported (Task 25) that searching a single character like "o"
  1049→      // returned only movies STARTING with "o" (e.g. "Ocean's Eleven",
  1050→      // "Once Upon a Time") but NOT movies CONTAINING "o" anywhere in the
  1051→      // title (e.g. "Thor", "Iron Man 2", "Doctor Strange"). Meanwhile
  1052→      // searching "ON" returned some movies with "on" anywhere — confusing.
  1053→      //
  1054→      // Root cause:
  1055→      //   - Strategy 1 (prefix search) only matches titles that START with
  1056→      //     the query. For "o", it returns plenty of results (Ocean's, Once,
  1057→      //     Oldboy, etc.), so prefixMovies is NOT empty.
  1058→      //   - The previous version of this Strategy 1.5 only fired when
  1059→      //     `prefixMovies.isEmpty && keywordResults.isEmpty` — which rarely
  1060→      //     happens for single letters (because Strategy 1 almost always
  1061→      //     finds prefix matches). So the substring fallback never ran,
  1062→      //     and the early-exit returned only the prefix matches.
  1063→      //   - For "ON": Strategy 1 returns fewer prefix matches (movies
  1064→      //     starting with "on" are rarer), so the substring fallback DID
  1065→      //     fire and found movies containing "on" anywhere. That's why
  1066→      //     Bro saw "some movies" for "ON" but "only starting-with-O
  1067→      //     movies" for "O".
  1068→      //
  1069→      // FIX (Task 25): Always run the substring fallback for short queries
  1070→      // (length <= 2), regardless of whether Strategies 1 and 2 returned
  1071→      // matches. The early-exit logic below already merges
  1072→      // prefixMovies + keywordResults + substringResults and applies a
  1073→      // client-side `title.contains(query)` filter for short queries, so
  1074→      // the user will see BOTH prefix matches (movies starting with "o")
  1075→      // AND substring matches (movies containing "o" anywhere) — exactly
  1076→      // what Bro wants.
  1077→      //
  1078→      // Cost: +1 Firestore query (~fetchLimit reads) for EVERY short
  1079→      // search. At fetchLimit=60 reads/query (limit=20 → 20*3 clamped to
  1080→      // 60) and Bro's typical usage, this adds ~5-10% to Firebase Reads
  1081→      // — acceptable per Bro's explicit OK.
  1082→      final substringResults = <Movie>[];
  1083→      if (lowerKeyword.length <= 2) {
  1084→        try {
  1085→          final subSnapshot = await _moviesRef.limit(fetchLimit).get();
  1086→          for (final doc in subSnapshot.docs) {
  1087→            final movie = Movie.fromMap(
  1088→              doc.data() as Map<String, dynamic>,
  1089→              docId: doc.id,
  1090→            );
  1091→            // Case-insensitive substring match on title
  1092→            if (movie.titleLowercase.contains(lowerKeyword)) {
  1093→              substringResults.add(movie);
  1094→            }
  1095→          }
  1096→        } catch (e) {
  1097→          debugPrint('_searchWithKeyword substring fallback failed: $e');
  1098→        }
  1099→      }
  1100→
  1101→      // Pre-combine what we have so far. If primary queries already returned
  1102→      // movies that match every name token AND year token, we can SKIP the
  1103→      // expensive broader fallback queries below (saves ~200 reads).
  1104→      final earlySeenIds = <String>{};
  1105→      final earlyAllMovies = <Movie>[];
  1106→      for (final m in [...prefixMovies, ...keywordResults, ...substringResults]) {
  1107→        if (!earlySeenIds.contains(m.id)) {
  1108→          earlySeenIds.add(m.id);
  1109→          earlyAllMovies.add(m);
  1110→        }
  1111→      }
  1112→      final earlyFiltered = earlyAllMovies.where((m) {
  1113→        final lowerTitle = m.titleLowercase;
  1114→        // For short single-token queries (<=2 chars), use substring match
  1115→        // instead of token-every-contains. This prevents "o" from being
  1116→        // rejected just because it isn't a full word in the title.
  1117→        final nameMatch = nameTokens.isEmpty ||
  1118→            (lowerKeyword.length <= 2 && nameTokens.length == 1
  1119→                ? lowerTitle.contains(nameTokens.first)
  1120→                : nameTokens.every((token) => lowerTitle.contains(token)));
  1121→        final yearMatch = yearTokens.isEmpty ||
  1122→            (m.year != null && yearTokens.contains(m.year!.toLowerCase()));
  1123→        return nameMatch && yearMatch;
  1124→      }).toList();
  1125→
  1126→      // --- EARLY EXIT: skip broader fallbacks if primary returned matches ---
  1127→      // Apply the same downstream filters + sort so the user sees the same
  1128→      // result shape they would have seen with the full query path.
  1129→      if (earlyFiltered.isNotEmpty) {
  1130→        var filtered = earlyFiltered;
  1131→        if (genre != null && genre.isNotEmpty) {
  1132→          filtered = filtered.where((m) => m.categories.contains(genre)).toList();
  1133→        }
  1134→        if (type != null && type.isNotEmpty) {
  1135→          filtered = filtered.where((m) => m.type == type).toList();
  1136→        }
  1137→        if (year != null && year.isNotEmpty) {
  1138→          filtered = filtered.where((m) => m.year == year).toList();
  1139→        }
  1140→        if (rating != null && rating.isNotEmpty) {
  1141→          final minRating = double.tryParse(rating) ?? 0.0;
  1142→          filtered = filtered.where((m) {
  1143→            final movieRating = double.tryParse(m.rating ?? '0') ?? 0.0;
  1144→            return movieRating >= minRating;
  1145→          }).toList();
  1146→        }
  1147→        // Sort results (same logic as the main path below).
  1148→        if (sortBy == 'rating') {
  1149→          filtered.sort((a, b) =>
  1150→              (double.tryParse(b.rating ?? '0') ?? 0.0)
  1151→              .compareTo(double.tryParse(a.rating ?? '0') ?? 0.0));
  1152→        } else if (sortBy == 'name') {
  1153→          filtered.sort((a, b) => a.title.toLowerCase().compareTo(b.title.toLowerCase()));
  1154→        } else {
  1155→          filtered.sort((a, b) {
  1156→            final aDate = b.updatedAt ?? b.createdAt ?? DateTime(2000);
  1157→            final bDate = a.updatedAt ?? a.createdAt ?? DateTime(2000);
  1158→            return aDate.compareTo(bDate);
  1159→          });
  1160→        }
  1161→        return {
  1162→          'movies': filtered.take(limit).toList(),
  1163→          'hasMore': filtered.length > limit,
  1164→          'lastDoc': null,
  1165→        };
  1166→      }
  1167→
  1168→      // --- FALLBACK: only when primary queries returned 0 matches ----------
  1169→      // Run broader queries to catch LEGACY movies missing both
  1170→      // 'title_lowercase' and 'search_keywords' fields.
  1171→      //
  1172→      // IMPORTANT: We use TWO parallel broader queries here instead of one:
  1173→      //   - broaderByUpdatedAt: orderBy('updatedAt', descending) — catches
  1174→      //     recently-updated movies that have 'updatedAt' field (added via
  1175→      //     addMovie / updateMovie).
  1176→      //   - broaderNoOrderBy:   plain .limit() without orderBy — catches
  1177→      //     LEGACY movies that have neither 'updatedAt' nor 'createdAt'
  1178→      //     field. Firestore's orderBy(field) silently excludes any doc
  1179→      //     missing that field — that's why a previous version of this
  1180→      //     code used orderBy('createdAt') and could NOT find movies that
  1181→      //     Bro had imported via an older code path that didn't set
  1182→      //     'createdAt'. Bro reported "search any movie → nothing shows"
  1183→      //     because of this exact bug. The no-orderBy path guarantees
  1184→      //     every movie in the collection is a candidate for the
  1185→      //     client-side title filter below.
  1186→      //
  1187→      // Both queries are limited to fetchLimit (60-200 docs). Pagination
  1188→      // via startAfterDocument only works on the orderBy path; the
  1189→      // no-orderBy path is best-effort and skipped when paginating.
  1190→      Query broaderByUpdatedAt = _moviesRef
  1191→          .orderBy('updatedAt', descending: true);
  1192→      if (startAfter != null) {
  1193→        broaderByUpdatedAt = broaderByUpdatedAt.startAfterDocument(startAfter);
  1194→      }
  1195→      broaderByUpdatedAt = broaderByUpdatedAt.limit(fetchLimit);
  1196→
  1197→      final broaderMovies = <Movie>[];
  1198→      try {
  1199→        final snapshot = await broaderByUpdatedAt.get();
  1200→        for (final doc in snapshot.docs) {
  1201→          broaderMovies.add(Movie.fromMap(
  1202→            doc.data() as Map<String, dynamic>,
  1203→            docId: doc.id,
  1204→          ));
  1205→        }
  1206→      } catch (e) {
  1207→        debugPrint('_searchWithKeyword broaderByUpdatedAt failed: $e — '
  1208→            'will rely on no-orderBy fallback');
  1209→      }
  1210→
  1211→      // No-orderBy fallback: only on first page (startAfter == null).
  1212→      // On subsequent pages, we'd need a stable cursor, which orderBy
  1213→      // provides — without orderBy, pagination is unsafe. So we accept
  1214→      // that legacy movies are only fully searchable on page 1.
  1215→      if (startAfter == null) {
  1216→        try {
  1217→          final legacySnapshot = await _moviesRef.limit(fetchLimit).get();
  1218→          for (final doc in legacySnapshot.docs) {
  1219→            final movie = Movie.fromMap(
  1220→              doc.data() as Map<String, dynamic>,
  1221→              docId: doc.id,
  1222→            );
  1223→            // Dedup against the orderBy results — same movie could appear
  1224→            // in both if it has updatedAt.
  1225→            final id = movie.id;
  1226→            final alreadySeen = broaderMovies.any((m) => m.id == id);
  1227→            if (!alreadySeen) {
  1228→              broaderMovies.add(movie);
  1229→            }
  1230→          }
  1231→        } catch (e) {
  1232→          debugPrint('_searchWithKeyword legacy fallback failed: $e');
  1233→        }
  1234→      }
  1235→
  1236→      // Combine results from all strategies, deduplicating by ID
  1237→      final seenIds = <String>{};
  1238→      final allMovies = <Movie>[];
  1239→
  1240→      for (final m in [...prefixMovies, ...keywordResults, ...substringResults, ...broaderMovies]) {
  1241→        if (!seenIds.contains(m.id)) {
  1242→          seenIds.add(m.id);
  1243→          allMovies.add(m);
  1244→        }
  1245→      }
  1246→
  1247→      // Apply token-based advanced filtering
  1248→      var filtered = allMovies.where((m) {
  1249→        // All name tokens must be contained in the movie title (case-insensitive)
  1250→        final lowerTitle = m.titleLowercase;
  1251→        final nameMatch = nameTokens.isEmpty ||
  1252→            nameTokens.every((token) => lowerTitle.contains(token));
  1253→
  1254→        // If year tokens present, at least one must match the movie year
  1255→        final yearMatch = yearTokens.isEmpty ||
  1256→            (m.year != null && yearTokens.contains(m.year!.toLowerCase()));
  1257→
  1258→        return nameMatch && yearMatch;
  1259→      }).toList();
  1260→
  1261→      // Apply additional filters
  1262→      if (genre != null && genre.isNotEmpty) {
  1263→        filtered = filtered.where((m) => m.categories.contains(genre)).toList();
  1264→      }
  1265→      if (type != null && type.isNotEmpty) {
  1266→        filtered = filtered.where((m) => m.type == type).toList();
  1267→      }
  1268→      if (year != null && year.isNotEmpty) {
  1269→        filtered = filtered.where((m) => m.year == year).toList();
  1270→      }
  1271→      if (rating != null && rating.isNotEmpty) {
  1272→        final minRating = double.tryParse(rating) ?? 0.0;
  1273→        filtered = filtered.where((m) {
  1274→          final movieRating = double.tryParse(m.rating ?? '0') ?? 0.0;
  1275→          return movieRating >= minRating;
  1276→        }).toList();
  1277→      }
  1278→
  1279→      // Sort results
  1280→      if (sortBy == 'rating') {
  1281→        filtered.sort((a, b) =>
  1282→            (double.tryParse(b.rating ?? '0') ?? 0.0)
  1283→            .compareTo(double.tryParse(a.rating ?? '0') ?? 0.0));
  1284→      } else if (sortBy == 'name') {
  1285→        filtered.sort((a, b) => a.title.toLowerCase().compareTo(b.title.toLowerCase()));
  1286→      } else {
  1287→        // 'latest' — sort by updatedAt with fallback to createdAt, so that
  1288→        // movies missing one of the timestamp fields still sort correctly
  1289→        // relative to the others (DateTime(2000) sentinel sorts to bottom).
  1290→        filtered.sort((a, b) {
  1291→          final aDate = b.updatedAt ?? b.createdAt ?? DateTime(2000);
  1292→          final bDate = a.updatedAt ?? a.createdAt ?? DateTime(2000);
  1293→          return aDate.compareTo(bDate);
  1294→        });
  1295→      }
  1296→
  1297→      final hasMore = filtered.length > limit ||
  1298→          broaderMovies.length >= fetchLimit;
  1299→
  1300→      // lastDoc: cursor for pagination. We can only paginate using the
  1301→      // orderBy('updatedAt') query, so we look up the last movie in our
  1302→      // combined list and find the corresponding snapshot. Since we don't
  1303→      // keep the snapshot around, we return null here and let the caller
  1304→      // re-run the search with the next "page" by skipping already-seen IDs.
  1305→      // For the search screen, infinite scroll beyond page 1 is rarely used
  1306→      // (users typically refine their query instead), so this is acceptable.
  1307→      return {
  1308→        'movies': filtered.take(limit).toList(),
  1309→        'hasMore': hasMore,
  1310→        'lastDoc': null,
  1311→      };
  1312→    } catch (e) {
  1313→      debugPrint('_searchWithKeyword failed: $e');
  1314→      return {
  1315→        'movies': <Movie>[],
  1316→        'hasMore': false,
  1317→        'lastDoc': null,
  1318→      };
  1319→    }
  1320→  }

  1631→  /// write. For a 100-movie batch import that's 400 fewer writes per run.
