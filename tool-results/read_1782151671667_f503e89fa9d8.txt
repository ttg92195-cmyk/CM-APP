    97→    97→  // Fetch the real app version (e.g. "1.9.0+15") from pubspec at runtime
    98→    98→  Future<void> _loadAppVersion() async {
    99→    99→    try {
   100→   100→      final info = await PackageInfo.fromPlatform();
   101→   101→      final version = info.version;
   102→   102→      final build = info.buildNumber;
   103→   103→      final formatted = build.isEmpty ? version : '$version+$build';
   104→   104→      if (mounted) {
   105→   105→        setState(() => _appVersion = formatted);
   106→   106→      }
   107→   107→    } catch (_) {
   108→   108→      // Fallback: leave empty so UI can fall back to a static label
   109→   109→    }
   110→   110→  }

   112→   112→  /// Silently refresh the Home tab's data. Called after returning from
   113→   113→  /// content-modifying screens (TMDB Generator, Admin Panel, etc.) so that
   114→   114→  /// newly-added or edited posts appear at the top of the Home feed
   115→   115→  /// without requiring the user to pull-to-refresh.
   116→   116→  ///
   117→   117→  /// Safe to call when Home is not the current tab — IndexedStack keeps
   118→   118→  /// HomeScreen alive, so the GlobalKey state is always available.
   119→   119→  void _refreshHomeIfMounted() {
   120→   120→    final state = _homeKey.currentState;
   121→   121→    if (state != null) {
   122→   122→      // ignore: avoid_dynamic_calls — HomeScreen.refresh() is a public API
   123→   123→      (state as dynamic).refresh();
   124→   124→    }
   125→   125→  }

   241→   241→  Widget _buildBottomNav(AppConfig appConfig, ThemeData theme) {
   242→   242→    final isDark = theme.brightness == Brightness.dark;
   243→   243→    final unselectedColor = isDark ? Colors.grey.shade500 : Colors.grey.shade600;
   244→   244→    const selectedColor = Color(0xFFE50914);
   245→   245→
   246→   246→    return Container(
   247→   247→      decoration: BoxDecoration(
   248→   248→        color: isDark ? const Color(0xFF121212) : Colors.white,
   249→   249→        boxShadow: [
   250→   250→          BoxShadow(
   251→   251→            color: isDark ? Colors.black26 : Colors.grey.shade300,
   252→   252→            blurRadius: 4,
   253→   253→            spreadRadius: 0,
   254→   254→            offset: const Offset(0, -1),
   255→   255→          ),
   256→   256→        ],
   257→   257→      ),
   258→   258→      child: NavigationBarTheme(
   259→   259→        data: NavigationBarThemeData(
   260→   260→          backgroundColor: isDark ? const Color(0xFF121212) : Colors.white,
   261→   261→          indicatorColor: Colors.transparent,
   262→   262→          height: 64,
   263→   263→          labelBehavior: NavigationDestinationLabelBehavior.alwaysShow,
   264→   264→          surfaceTintColor: Colors.transparent,
   265→   265→          overlayColor: WidgetStateProperty.all(Colors.transparent),
   266→   266→          iconTheme: WidgetStateProperty.resolveWith<IconThemeData?>((states) {
   267→   267→            final isSelected = states.contains(WidgetState.selected);
   268→   268→            return IconThemeData(
   269→   269→              size: 24,
   270→   270→              color: isSelected ? selectedColor : unselectedColor,
   271→   271→            );
   272→   272→          }),
   273→   273→          labelTextStyle: WidgetStateProperty.resolveWith<TextStyle?>((states) {
   274→   274→            final isSelected = states.contains(WidgetState.selected);
   275→   275→            return TextStyle(
   276→   276→              fontSize: 11,
   277→   277→              fontWeight: isSelected ? FontWeight.w600 : FontWeight.normal,
   278→   278→              color: isSelected ? selectedColor : unselectedColor,
   279→   279→            );
   280→   280→          }),
   281→   281→        ),
   282→   282→        child: NavigationBar(
   283→   283→        selectedIndex: _currentIndex,
   284→   284→        onDestinationSelected: (index) {
   285→   285→          // If user taps Home tab while already on Home, treat it as a
   286→   286→          // refresh trigger (silently — no skeleton). This mirrors the
   287→   287→          // common "tap the active tab to refresh" pattern in apps like
   288→   288→          // Twitter / Instagram. Otherwise just switch tabs.
   289→   289→          if (index == _currentIndex && index == kHomeTab) {
   290→   290→            _refreshHomeIfMounted();
   291→   291→            return;
   292→   292→          }
   293→   293→          setState(() {
   294→   294→            _currentIndex = index;
   295→   295→          });
   296→   296→          // Trigger data refresh when switching to Movies/Series tabs
   297→   297→          if (index == kMoviesTab) {
   298→   298→            final state = _moviesKey.currentState;
   299→   299→            if (state != null) (state as dynamic).onTabSelected();
   300→   300→          } else if (index == kSeriesTab) {
   301→   301→            final state = _seriesKey.currentState;
   302→   302→            if (state != null) (state as dynamic).onTabSelected();
   303→   303→          } else if (index == kHomeTab) {
   304→   304→            // Returning to Home — silently refresh in case the user
   305→   305→            // added/edited content elsewhere since Home was last visible.
   306→   306→            _refreshHomeIfMounted();
   307→   307→          }
   308→   308→        },
   309→   309→        destinations: [
   310→   310→          NavigationDestination(
   311→   311→            icon: const Icon(Icons.home_outlined),
   312→   312→            selectedIcon: const Icon(Icons.home),
   313→   313→            label: appConfig.translate('home'),
   314→   314→          ),
   315→   315→          NavigationDestination(
   316→   316→            icon: const Icon(Icons.movie_outlined),
   317→   317→            selectedIcon: const Icon(Icons.movie),
   318→   318→            label: appConfig.translate('movies'),
   319→   319→          ),
   320→   320→          NavigationDestination(
   321→   321→            icon: const Icon(Icons.tv_outlined),
   322→   322→            selectedIcon: const Icon(Icons.tv),
   323→   323→            label: appConfig.translate('series'),
   324→   324→          ),
   325→   325→          NavigationDestination(
   326→   326→            icon: const Icon(Icons.settings_outlined),
   327→   327→            selectedIcon: const Icon(Icons.settings),
   328→   328→            label: appConfig.translate('settings'),
   329→   329→          ),
   330→   330→        ],
   331→   331→      ),
   332→   332→      ),
   333→   333→    );
   334→   334→  }
