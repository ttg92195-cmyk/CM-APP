   203→  /// Handle connectivity changes in real-time
   204→  /// Shows "No Internet" screen immediately when connection is lost,
   205→  /// and auto-refreshes when connection is restored
   206→  void _onConnectivityChanged(List<ConnectivityResult> results) {
   207→    final hasConnection = results.any((r) => r != ConnectivityResult.none);
   208→
   209→    if (!hasConnection && _hasInternet) {
   210→      // Internet just went down — show No Internet screen
   211→      if (mounted) {
   212→        setState(() => _hasInternet = false);
   213→      }
   214→    } else if (hasConnection && !_hasInternet) {
   215→      // Internet just came back — auto-refresh
   216→      if (mounted) {
   217→        setState(() {
   218→          _hasInternet = true;
   219→          _checkingInternet = false;
   220→        });
   221→        // Check for updates now that we have internet
   222→        _checkForUpdate();
   223→      }
   224→    }
   225→  }

   231→  // Feature 1: Check internet connection using DNS lookup
   232→  Future<void> _checkInternet() async {
   233→    if (_checkingInternet) return;
   234→    setState(() => _checkingInternet = true);
   235→    try {
   236→      final result = await InternetAddress.lookup('google.com');
   237→      final hasConnection = result.isNotEmpty && result[0].rawAddress.isNotEmpty;
   238→      if (mounted) {
   239→        setState(() {
   240→          _hasInternet = hasConnection;
   241→          _checkingInternet = false;
   242→        });
   243→        // If we have internet, check for force update
   244→        if (hasConnection) {
   245→          _checkForUpdate();
   246→        }
   247→      }
   248→    } catch (_) {
   249→      if (mounted) {
   250→        setState(() {
   251→          _hasInternet = false;
   252→          _checkingInternet = false;
   253→        });
   254→      }
   255→    }
   256→  }
   257→
   258→  // Feature 2: Check for force update from Firestore config
   259→  Future<void> _checkForUpdate() async {
   260→    try {
   261→      final doc = await FirebaseFirestore.instance
   262→          .collection('config')
   263→          .doc('app_version')
   264→          .get();
   265→      if (!doc.exists) return;
   266→      final data = doc.data()!;
   267→      _latestVersion = data['latestVersion'] as String? ?? '';
   268→      final forceUpdate = data['forceUpdate'] as bool? ?? false;
   269→
   270→      if (_latestVersion.isNotEmpty && forceUpdate) {
   271→        const currentVersion = '1.9.0';
   272→        if (_isNewerVersion(_latestVersion, currentVersion)) {
   273→          if (mounted) {
   274→            setState(() => _forceUpdate = true);
   275→          }
   276→        }
   277→      }
   278→    } catch (e) {
   279→      debugPrint('Version check error: $e');
   280→    }
   281→  }
   282→
   283→  // Feature 2: Compare version strings (e.g. "2.0.0" > "1.9.0")
   284→  bool _isNewerVersion(String latest, String current) {
   285→    try {
   286→      final latestParts = latest.split('.').map(int.parse).toList();
   287→      final currentParts = current.split('.').map(int.parse).toList();
   288→      for (int i = 0; i < 3; i++) {
   289→        final l = i < latestParts.length ? latestParts[i] : 0;
   290→        final c = i < currentParts.length ? currentParts[i] : 0;
   291→        if (l > c) return true;
   292→        if (l < c) return false;
   293→      }
   294→      return false;
   295→    } catch (e) {
   296→      debugPrint('Version comparison error: $e');
   297→      return false;
   298→    }
   299→  }

   447→  // L3: Track app lifecycle for session timeout
   448→  @override
   449→  void didChangeAppLifecycleState(AppLifecycleState state) {
   450→    if (state == AppLifecycleState.resumed) {
   451→      try {
   452→        // Recover orphaned downloads (stuck in "downloading" after app was in background)
   453→        DownloadManagerService.instance.recoverOrphanedDownloads();
   454→
   455→        // App came back to foreground — check session timeout
   456→        final appConfig = Provider.of<AppConfig>(context, listen: false);
   457→        if (appConfig.isLoggedIn && appConfig.checkSessionTimeout()) {
   458→          // Session timed out while app was in background
   459→          // Only show SnackBar if still mounted (prevents crash)
   460→          if (mounted) {
   461→            ScaffoldMessenger.of(context).showSnackBar(
   462→              const SnackBar(
   463→                content: Text('Session expired due to inactivity. Please login again.'),
   464→                backgroundColor: Colors.orange,
   465→                duration: Duration(seconds: 4),
   466→              ),
   467→            );
   468→          }
   469→        } else {
   470→          appConfig.recordActivity();
   471→        }
   472→      } catch (e) {
   473→        debugPrint('App resume lifecycle error: $e');
   474→        // Don't crash — just continue normally
   475→      }
   476→    }
   477→  }

   490→
   491→    // IMPORTANT: We use a Listener (NOT GestureDetector) at the root so that
   492→    // activity tracking observes pointer events WITHOUT participating in the
   493→    // gesture arena. A GestureDetector with both onTap + onPanDown at the
   494→    // root would force the arena to disambiguate tap-vs-pan on every touch,
   495→    // which competes with the internal drag recognizer used by EditableText's
   496→    // text-selection handles — causing the "selection handle drag lock" bug
   497→    // (handles visible but cannot be dragged to extend/reduce selection) on
   498→    // every TextField / TextFormField / SelectableText in the entire app.
   499→    //
   500→    // Listener simply observes raw pointer events; it never claims gestures,
   501→    // so selection handles, scrolls, taps, and all other interactions work
   502→    // normally while we still get an activity tick on every user touch.
   503→    return Listener(
   504→      // L3: Record activity on any user pointer-down to keep session alive.
   505→      // Every user interaction (tap, drag, scroll, selection-handle drag,
   506→      // long-press, etc.) starts with a pointer-down event, so this single
   507→      // callback covers all interaction types without interfering with them.
   508→      onPointerDown: (_) => appConfig.recordActivity(),
   509→      onPointerMove: (_) => appConfig.recordActivity(),
   510→      child: MaterialApp(
   511→        title: 'KMM',
   512→        debugShowCheckedModeBanner: false,
   513→        navigatorKey: navigatorKey,

   613→      //
   614→      // Bro's UI/UX concern: Myanmar text is taller than English and uses
   615→      // combining marks (stacked diacritics) that often get clipped when
   616→      // line-height is too tight. Two mitigations here:
   617→      //
   618→      // 1. fontFamilyFallback: ['Noto Sans Myanmar', 'Padauk']
   619→      //    On Android, both fonts are typically pre-installed at the system
   620→      //    level (Noto Sans Myanmar ships with every Android 7+ device via
   621→      //    the system font stack). On devices that don't have them, Flutter
   622→      //    falls back to Roboto, which still has Myanmar glyph coverage via
   623→      //    its bundled Noto fallback.
   624→      //
   625→      // 2. height: 1.3 (line-height multiplier)
   626→      //    Gives Myanmar stacked diacritics breathing room. For English
   627→      //    text this is slightly loose but doesn't look weird — it's a
   628→      //    safe default that works for both scripts.
   629→      //
   630→      // 3. fontSize base untouched — Material 3 defaults are fine.

   632→      //    handle text-overflow (Phase 2 work, separate task).
