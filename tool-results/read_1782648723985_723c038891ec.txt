     7→/// Admin Notification Page — allows admins to compose and send push notifications
     8→/// to all users, and view notification history.
     9→///
    10→/// Flow:
    11→/// 1. Admin fills in Title, Body, and optional Movie ID/Slug
    12→/// 2. Taps "Send Notification"
    13→/// 3. Notification is saved to Firestore `notifications` collection
    14→/// 4. A Cloud Function triggers on the new document and sends FCM push to `movies_all` topic
    15→/// 5. All subscribed devices receive the push notification
    16→class AdminNotificationPage extends StatefulWidget {
    17→  const AdminNotificationPage({super.key});

    42→  /// Send notification: saves to Firestore and triggers Cloud Function push
    43→  Future<void> _sendNotification() async {
    44→    if (!_formKey.currentState!.validate()) return;
    45→
    46→    final confirmed = await showDialog<bool>(
    47→      context: context,
    48→      builder: (ctx) => AlertDialog(
    49→        title: const Text('Send Notification'),
    50→        content: Column(
    51→          mainAxisSize: MainAxisSize.min,
    52→          crossAxisAlignment: CrossAxisAlignment.start,
    53→          children: [
    54→            Text('Send this notification to all users?'),
    55→            const SizedBox(height: 12),
    56→            Container(
    57→              padding: const EdgeInsets.all(12),
    58→              decoration: BoxDecoration(
    59→                color: const Color(0xFFE50914).withOpacity(0.1),
    60→                borderRadius: BorderRadius.circular(8),
    61→                border: Border.all(color: const Color(0xFFE50914).withOpacity(0.3)),
    62→              ),
    63→              child: Column(
    64→                crossAxisAlignment: CrossAxisAlignment.start,
    65→                children: [
    66→                  Text(
    67→                    _titleController.text.trim(),
    68→                    style: const TextStyle(fontWeight: FontWeight.bold, fontSize: 14),
    69→                  ),
    70→                  const SizedBox(height: 4),
    71→                  Text(
    72→                    _bodyController.text.trim(),
    73→                    style: const TextStyle(fontSize: 13),
    74→                    maxLines: 3,
    75→                    overflow: TextOverflow.ellipsis,
    76→                  ),
    77→                ],
    78→              ),
    79→            ),
    80→          ],
    81→        ),
    82→        actions: [
    83→          TextButton(
    84→            onPressed: () => Navigator.pop(ctx, false),
    85→            child: const Text('Cancel'),
    86→          ),
    87→          ElevatedButton.icon(
    88→            onPressed: () => Navigator.pop(ctx, true),
    89→            icon: const Icon(Icons.send, size: 18),
    90→            label: const Text('Send'),
    91→            style: ElevatedButton.styleFrom(
    92→              backgroundColor: const Color(0xFFE50914),
    93→              foregroundColor: Colors.white,
    94→            ),
    95→          ),
    96→        ],
    97→      ),
    98→    );
    99→
   100→    if (confirmed != true) return;
   101→
   102→    setState(() => _isSending = true);
   103→
   104→    try {
   105→      final user = FirebaseAuth.instance.currentUser;
   106→      if (user == null) throw Exception('Not authenticated');
   107→
   108→      final notification = NotificationModel(
   109→        id: '', // Firestore will auto-generate
   110→        title: _titleController.text.trim(),
   111→        body: _bodyController.text.trim(),
   112→        movieId: _includeMovieLink && _movieIdController.text.trim().isNotEmpty
   113→            ? _movieIdController.text.trim()
   114→            : null,
   115→        movieSlug: _includeMovieLink && _movieSlugController.text.trim().isNotEmpty
   116→            ? _movieSlugController.text.trim()
   117→            : null,
   118→        sentBy: user.email ?? user.uid,
   119→        sentAt: DateTime.now(),
   120→        isSent: false, // Will be updated after OneSignal confirms
   121→      );
   122→
   123→      // Save to Firestore for history tracking
   124→      final docRef = await FirebaseFirestore.instance
   125→          .collection('notifications')
   126→          .add(notification.toFirestore());
   127→
   128→      // Send push notification via OneSignal REST API (free, no server needed)
   129→      final result = await FcmNotificationService().sendNotificationToAll(
   130→        title: notification.title,
   131→        body: notification.body,
   132→        movieId: notification.movieId,
   133→        movieSlug: notification.movieSlug,
   134→      );
   135→
   136→      final sent = result['success'] == true;
   137→      final errorMsg = result['error'] as String?;
   138→      final recipients = result['recipients'] as int?;
   139→
   140→      // Update Firestore with send status
   141→      await FirebaseFirestore.instance
   142→          .collection('notifications')
   143→          .doc(docRef.id)
   144→          .update({'isSent': sent});
   145→
   146→      if (mounted) {
   147→        // Determine success message based on result
   148→        String message;
   149→        Color bgColor;
   150→
   151→        if (sent && recipients != null && recipients > 0) {
   152→          message = 'Notification sent to $recipients device(s)!';
   153→          bgColor = Colors.green;
   154→        } else if (sent && (recipients == null || recipients == 0)) {
   155→          // API accepted but no subscribers yet — this is normal for new apps
   156→          message = 'Notification queued! Subscribers will receive it once they open the app.';
   157→          bgColor = Colors.blue;
   158→        } else {
   159→          message = 'Push failed: ${errorMsg ?? "Unknown error"}';
   160→          bgColor = Colors.orange;
   161→        }
   162→
   163→        ScaffoldMessenger.of(context).showSnackBar(
   164→          SnackBar(
   165→            content: Text(message),
   166→            backgroundColor: bgColor,
   167→            behavior: SnackBarBehavior.floating,
   168→            duration: Duration(seconds: sent ? 4 : 6),
   169→          ),
   170→        );
   171→        // Clear form
   172→        _titleController.clear();
   173→        _bodyController.clear();
   174→        _movieIdController.clear();
   175→        _movieSlugController.clear();
   176→        setState(() {
   177→          _includeMovieLink = false;
   178→          _isSending = false;
   179→        });
   180→      }
   181→    } catch (e) {
   182→      if (mounted) {
   183→        setState(() => _isSending = false);
   184→        ScaffoldMessenger.of(context).showSnackBar(
   185→          SnackBar(
   186→            content: Text('Failed to send: ${e.toString()}'),
   187→            backgroundColor: Colors.red,
   188→            behavior: SnackBarBehavior.floating,
   189→          ),
   190→        );
   191→      }
   192→    }
   193→  }

   195→  /// Delete a notification from Firestore
   196→  Future<void> _deleteNotification(String notificationId) async {
   197→    final confirmed = await showDialog<bool>(
   198→      context: context,
   199→      builder: (ctx) => AlertDialog(
   200→        title: const Text('Delete Notification'),
   201→        content: const Text('Are you sure you want to delete this notification?'),
   202→        actions: [
   203→          TextButton(
   204→            onPressed: () => Navigator.pop(ctx, false),
   205→            child: const Text('Cancel'),
   206→          ),
   207→          TextButton(
   208→            onPressed: () => Navigator.pop(ctx, true),
   209→            style: TextButton.styleFrom(foregroundColor: Colors.red),
   210→            child: const Text('Delete'),
   211→          ),
   212→        ],
   213→      ),
   214→    );
   215→
   216→    if (confirmed == true) {
   217→      try {
   218→        await FirebaseFirestore.instance
   219→            .collection('notifications')
   220→            .doc(notificationId)
   221→            .delete();
   222→        if (mounted) {
   223→          ScaffoldMessenger.of(context).showSnackBar(
   224→            const SnackBar(content: Text('Notification deleted')),
   225→          );
   226→        }
   227→      } catch (e) {
   228→        if (mounted) {
   229→          ScaffoldMessenger.of(context).showSnackBar(
   230→            SnackBar(
   231→              content: Text('Delete failed: $e'),
   232→              backgroundColor: Colors.red,
   233→            ),
   234→          );
   235→        }
   236→      }
   237→    }
   238→  }

   538→  Widget _buildHistoryTab(bool isDark) {
   539→    return StreamBuilder<QuerySnapshot>(
   540→      stream: FirebaseFirestore.instance
   541→          .collection('notifications')
   542→          .orderBy('sentAt', descending: true)
   543→          .limit(50)
   544→          .snapshots(),
   545→      builder: (context, snapshot) {
   546→        if (snapshot.connectionState == ConnectionState.waiting) {
   547→          return const Center(child: CircularProgressIndicator());
   548→        }
   549→
   550→        if (snapshot.hasError) {
   551→          return Center(
   552→            child: Column(
   553→              mainAxisAlignment: MainAxisAlignment.center,
   554→              children: [
   555→                const Icon(Icons.error_outline, size: 48, color: Colors.redAccent),
   556→                const SizedBox(height: 16),
   557→                Text('Error: ${snapshot.error}'),
   558→              ],
   559→            ),
   560→          );
   561→        }
   562→
   563→        final docs = snapshot.data?.docs ?? [];
   564→
   565→        if (docs.isEmpty) {
   566→          return Center(
   567→            child: Column(
   568→              mainAxisAlignment: MainAxisAlignment.center,
   569→              children: [
   570→                Icon(
   571→                  Icons.notifications_off_outlined,
   572→                  size: 64,
   573→                  color: isDark ? Colors.white24 : Colors.grey.shade400,
   574→                ),
   575→                const SizedBox(height: 16),
   576→                Text(
   577→                  'No notifications sent yet',
   578→                  style: TextStyle(
   579→                    fontSize: 16,
   580→                    color: isDark ? Colors.white54 : Colors.black54,
   581→                  ),
   582→                ),
   583→                const SizedBox(height: 8),
   584→                Text(
   585→                  'Compose and send your first notification',
   586→                  style: TextStyle(
   587→                    fontSize: 13,
   588→                    color: isDark ? Colors.white38 : Colors.black38,
   589→                  ),
   590→                ),
   591→              ],
   592→            ),
   593→          );
   594→        }
   595→
   596→        return ListView.builder(
   597→          padding: const EdgeInsets.symmetric(horizontal: 12, vertical: 8),
   598→          itemCount: docs.length,
   599→          itemBuilder: (context, index) {
   600→            final notification = NotificationModel.fromFirestore(docs[index]);
   601→            return _buildNotificationHistoryItem(notification, isDark);
   602→          },
   603→        );
   604→      },
   605→    );
   606→  }
