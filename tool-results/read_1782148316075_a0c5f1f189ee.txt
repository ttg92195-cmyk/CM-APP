   185→  /// Handle connectivity changes in real-time
   186→  /// Shows "No Internet" screen immediately when connection is lost,
   187→  /// and auto-refreshes when connection is restored
   188→  void _onConnectivityChanged(List<ConnectivityResult> results) {
   189→    final hasConnection = results.any((r) => r != ConnectivityResult.none);
   190→
   191→    if (!hasConnection && _hasInternet) {
   192→      // Internet just went down — show No Internet screen
   193→      if (mounted) {
   194→        setState(() => _hasInternet = false);
   195→      }
   196→    } else if (hasConnection && !_hasInternet) {
   197→      // Internet just came back — auto-refresh
   198→      if (mounted) {
   199→        setState(() {
   200→          _hasInternet = true;
   201→          _checkingInternet = false;
   202→        });
   203→        // Check for updates now that we have internet
   204→        _checkForUpdate();
   205→      }
   206→    }
   207→  }

   213→  // Feature 1: Check internet connection using DNS lookup
   214→  Future<void> _checkInternet() async {
   215→    if (_checkingInternet) return;
   216→    setState(() => _checkingInternet = true);
   217→    try {
   218→      final result = await InternetAddress.lookup('google.com');
   219→      final hasConnection = result.isNotEmpty && result[0].rawAddress.isNotEmpty;
   220→      if (mounted) {
   221→        setState(() {
   222→          _hasInternet = hasConnection;
   223→          _checkingInternet = false;
   224→        });
   225→        // If we have internet, check for force update
   226→        if (hasConnection) {
   227→          _checkForUpdate();
   228→        }
   229→      }
   230→    } catch (_) {
   231→      if (mounted) {
   232→        setState(() {
   233→          _hasInternet = false;
   234→          _checkingInternet = false;
   235→        });
   236→      }
   237→    }
   238→  }

   240→  // Feature 2: Check for force update from Firestore config
   241→  Future<void> _checkForUpdate() async {
   242→    try {
   243→      final doc = await FirebaseFirestore.instance
   244→          .collection('config')
   245→          .doc('app_version')
   246→          .get();
   247→      if (!doc.exists) return;
   248→      final data = doc.data()!;
   249→      _latestVersion = data['latestVersion'] as String? ?? '';
   250→      final forceUpdate = data['forceUpdate'] as bool? ?? false;
   251→
   252→      if (_latestVersion.isNotEmpty && forceUpdate) {
   253→        const currentVersion = '1.9.0';
   254→        if (_isNewerVersion(_latestVersion, currentVersion)) {
   255→          if (mounted) {
   256→            setState(() => _forceUpdate = true);
   257→          }
   258→        }
   259→      }
   260→    } catch (e) {
   261→      debugPrint('Version check error: $e');
   262→    }
   263→  }
   264→
   265→  // Feature 2: Compare version strings (e.g. "2.0.0" > "1.9.0")
   266→  bool _isNewerVersion(String latest, String current) {
   267→    try {
   268→      final latestParts = latest.split('.').map(int.parse).toList();
   269→      final currentParts = current.split('.').map(int.parse).toList();
   270→      for (int i = 0; i < 3; i++) {
   271→        final l = i < latestParts.length ? latestParts[i] : 0;
   272→        final c = i < currentParts.length ? currentParts[i] : 0;
   273→        if (l > c) return true;
   274→        if (l < c) return false;
   275→      }
   276→      return false;
   277→    } catch (e) {
   278→      debugPrint('Version comparison error: $e');
   279→      return false;
   280→    }
   281→  }

   430→  @override
   431→  void didChangeAppLifecycleState(AppLifecycleState state) {
   432→    if (state == AppLifecycleState.resumed) {
   433→      try {
   434→        // Recover orphaned downloads (stuck in "downloading" after app was in background)

   437→        // App came back to foreground — check session timeout
   438→        final appConfig = Provider.of<AppConfig>(context, listen: false);
   439→        if (appConfig.isLoggedIn && appConfig.checkSessionTimeout()) {
   440→          // Session timed out while app was in background
   441→          // Only show SnackBar if still mounted (prevents crash)
   442→          if (mounted) {

   446→                backgroundColor: Colors.orange,

   451→        } else {
   452→          appConfig.recordActivity();
   453→        }
   454→      } catch (e) {
   455→        debugPrint('App resume lifecycle error: $e');
   456→        // Don't crash — just continue normally
   457→      }
   458→    }

   472→
   473→    // IMPORTANT: We use a Listener (NOT GestureDetector) at the root so that
   474→    // activity tracking observes pointer events WITHOUT participating in the
   475→    // gesture arena. A GestureDetector with both onTap + onPanDown at the
   476→    // root would force the arena to disambiguate tap-vs-pan on every touch,
   477→    // which competes with the internal drag recognizer used by EditableText's
   478→    // text-selection handles — causing the "selection handle drag lock" bug
   479→    // (handles visible but cannot be dragged to extend/reduce selection) on
   480→    // every TextField / TextFormField / SelectableText in the entire app.
   481→    //
   482→    // Listener simply observes raw pointer events; it never claims gestures,
   483→    // so selection handles, scrolls, taps, and all other interactions work
   484→    // normally while we still get an activity tick on every user touch.
   485→    return Listener(
   486→      // L3: Record activity on any user pointer-down to keep session alive.
   487→      // Every user interaction (tap, drag, scroll, selection-handle drag,
   488→      // long-press, etc.) starts with a pointer-down event, so this single
   489→      // callback covers all interaction types without interfering with them.

   491→      onPointerMove: (_) => appConfig.recordActivity(),

   493→        title: 'KMM',

   588→  ThemeData _buildDarkTheme() {
   589→    final base = ThemeData.dark(useMaterial3: true);
   590→    return base.copyWith(
   591→      // Task 32: Myanmar-aware text theme.
   592→      //
   593→      // Bro's UI/UX concern: Myanmar text is taller than English and uses
   594→      // combining marks (stacked diacritics) that often get clipped when
   595→      // line-height is too tight. Two mitigations here:
   596→      //
   597→      // 1. fontFamilyFallback: ['Noto Sans Myanmar', 'Padauk']
   598→      //    On Android, both fonts are typically pre-installed at the system
   599→      //    level (Noto Sans Myanmar ships with every Android 7+ device via
   600→      //    the system font stack). On devices that don't have them, Flutter
   601→      //    falls back to Roboto, which still has Myanmar glyph coverage via
   602→      //    its bundled Noto fallback.
   603→      //
   604→      // 2. height: 1.3 (line-height multiplier)
