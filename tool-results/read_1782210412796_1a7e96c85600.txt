    54→      List<TagAndGenres> genres = [];

    58→      // PARALLEL FETCH — previously sequential (genres → tags → collections),

    67→        }),

    69→          debugPrint('Error loading tags: $e');

   427→// ==================== FILTER RESULT PAGE (public) ====================
   428→class FilterResultPage extends StatefulWidget {
   429→  final String title;
   430→  final String? genreName;
   431→  final String? tagName;
   432→  final String? collectionName;
   433→  final String? typeFilter; // 'movie' or 'series'
   434→
   435→  const FilterResultPage({
   436→    super.key,
   437→    required this.title,
   438→    this.genreName,
   439→    this.tagName,
   440→    this.collectionName,
   441→    this.typeFilter,
   442→  });
   443→
   444→  @override
   445→  State<FilterResultPage> createState() => _FilterResultPageState();
   446→}
   447→
   448→class _FilterResultPageState extends State<FilterResultPage> {
   449→  final FirestoreContentService _contentService = FirestoreContentService();
   450→
   451→  List<Movie> _movies = [];
   452→  List<Movie> _filteredMovies = [];
   453→  DocumentSnapshot? _lastDoc;
   454→  bool _hasMore = true;
   455→  bool _isLoading = true;
   456→  bool _isLoadingMore = false;
   457→  // Track seen IDs to prevent duplicates
   458→  final Set<String> _seenIds = {};
   459→
   460→  @override
   461→  void initState() {
   462→    super.initState();
   463→    _loadMovies();
   464→  }
   465→
   466→  Future<void> _loadMovies() async {
   467→    setState(() {
   468→      _isLoading = true;
   469→      _seenIds.clear();
   470→    });
   471→    try {
   472→      Map<String, dynamic> result;
   473→      if (widget.genreName != null) {
   474→        result = await _contentService.getMoviesByGenre(
   475→          widget.genreName!, limit: 20, typeFilter: widget.typeFilter, // PAGINATION: 20/page (Task 38 Req 5)
   476→        );
   477→      } else if (widget.tagName != null) {
   478→        result = await _contentService.getMoviesByTag(
   479→          widget.tagName!, limit: 20, typeFilter: widget.typeFilter, // PAGINATION: 20/page (Task 38 Req 5)
   480→        );
   481→      } else if (widget.collectionName != null) {
   482→        result = await _contentService.getMoviesByCollection(
   483→          widget.collectionName!, limit: 20, typeFilter: widget.typeFilter, // PAGINATION: 20/page (Task 38 Req 5)
   484→        );
   485→      } else {
   486→        result = await _contentService.getMovies(limit: 20); // PAGINATION: 20/page
   487→      }
   488→
   489→      if (mounted) {
   490→        final allMovies = result['movies'] as List<Movie>;
   491→
   492→        // Track IDs to prevent future duplicates
   493→        for (final m in allMovies) {
   494→          _seenIds.add(m.id);
   495→        }
   496→
   497→        // Apply type filter if specified (Movies or Series sub-tab)
   498→        List<Movie> filtered;
   499→        if (widget.typeFilter != null) {
   500→          filtered = allMovies
   501→              .where((m) => m.type == widget.typeFilter)
   502→              .toList();
   503→        } else {
   504→          filtered = allMovies;
   505→        }
   506→
   507→        // NOTE: previously had an artificial 600ms skeleton floor here.
   508→        // Removed in Task 36 #2 — see movies_page.dart for the rationale.
   509→        setState(() {
   510→          _movies = allMovies;
   511→          _filteredMovies = filtered;
   512→          _hasMore = result['hasMore'] as bool;
   513→          _lastDoc = result['lastDoc'] as DocumentSnapshot?;
   514→          _isLoading = false;
   515→        });
   516→
   517→        // Task 38 Req 5 — post-frame auto-load safety net.
   518→        // If the first page returned fewer items than would fill the
   519→        // viewport (e.g., on a large tablet showing 30+ items per screen,
   520→        // or a genre with exactly 20 docs that doesn't fill the screen),
   521→        // the scroll listener never fires — and infinite scroll silently
   522→        // dies. Schedule a `_loadMore()` on the next frame; `_loadMore`
   523→        // will no-op if `_hasMore` is already false (e.g., the genre only
   524→        // had 20 docs total).
   525→        if (_hasMore && _filteredMovies.length < 30) {
   526→          WidgetsBinding.instance.addPostFrameCallback((_) {
   527→            if (mounted && _hasMore && !_isLoadingMore) {
   528→              _loadMore();
   529→            }
   530→          });
   531→        }
   532→      }
   533→    } catch (e) {
   534→      debugPrint('FilterResultPage load error: $e');
   535→      if (mounted) setState(() => _isLoading = false);
   536→    }
   537→  }
   538→
   539→  Future<void> _loadMore() async {
   540→    if (_isLoadingMore || !_hasMore) return;
   541→    setState(() => _isLoadingMore = true);
   542→    try {
   543→      Map<String, dynamic> result;
   544→      if (widget.genreName != null) {
   545→        result = await _contentService.getMoviesByGenre(
   546→          widget.genreName!, limit: 20, startAfter: _lastDoc,
   547→          typeFilter: widget.typeFilter, // Task 38 Req 5
   548→        );
   549→      } else if (widget.tagName != null) {
   550→        result = await _contentService.getMoviesByTag(
   551→          widget.tagName!, limit: 20, startAfter: _lastDoc,
   552→          typeFilter: widget.typeFilter, // Task 38 Req 5
   553→        );
   554→      } else if (widget.collectionName != null) {
   555→        result = await _contentService.getMoviesByCollection(
   556→          widget.collectionName!, limit: 20, startAfter: _lastDoc,
   557→          typeFilter: widget.typeFilter, // Task 38 Req 5
   558→        );
   559→      } else {
   560→        result = await _contentService.getMovies(limit: 20, startAfter: _lastDoc);
   561→      }
   562→
   563→      if (mounted) {
   564→        final newMovies = result['movies'] as List<Movie>;
   565→
   566→        // Deduplicate by ID to prevent duplicates
   567→        final dedupedMovies = <Movie>[];
   568→        for (final m in newMovies) {
   569→          if (!_seenIds.contains(m.id)) {
   570→            _seenIds.add(m.id);
   571→            dedupedMovies.add(m);
   572→          }
   573→        }
   574→
   575→        List<Movie> filtered;
   576→        if (widget.typeFilter != null) {
   577→          filtered = dedupedMovies
   578→              .where((m) => m.type == widget.typeFilter)
   579→              .toList();
   580→        } else {
   581→          filtered = dedupedMovies;
   582→        }
   583→
   584→        setState(() {
   585→          _movies.addAll(dedupedMovies);
   586→          _filteredMovies.addAll(filtered);
   587→          _hasMore = result['hasMore'] as bool && newMovies.isNotEmpty;
   588→          _lastDoc = result['lastDoc'] as DocumentSnapshot?;
   589→          _isLoadingMore = false;
   590→        });
   591→
   592→        // Task 38 Req 5 — chained auto-load safety net (mirror of the one
   593→        // in _loadMovies). If after a `_loadMore` we still have fewer
   594→        // than 30 visible items AND `_hasMore` is still true, the grid
   595→        // likely still doesn't fill the viewport — schedule another
   596→        // `_loadMore` on the next frame. Caps infinite recursion because
   597→        // `_loadMore` no-ops once `_hasMore` flips false (Firestore
   598→        // returned fewer than `limit` docs).
   599→        if (_hasMore && _filteredMovies.length < 30) {
   600→          WidgetsBinding.instance.addPostFrameCallback((_) {
   601→            if (mounted && _hasMore && !_isLoadingMore) {
   602→              _loadMore();
   603→            }
   604→          });
   605→        }
   606→      }
   607→    } catch (e) {
   608→      if (mounted) setState(() => _isLoadingMore = false);
   609→    }
   610→  }

   613→  Widget build(BuildContext context) {
   614→    final theme = Theme.of(context);
   615→
   616→    return Scaffold(
   617→      appBar: AppBar(
   618→        title: Text(
   619→          widget.title,
   620→          maxLines: 1,
   621→          overflow: TextOverflow.ellipsis,

   624→      body: _isLoading
   625→          ? _buildSkeletonGrid()
   626→          : RefreshIndicator(
   627→              onRefresh: _loadMovies,
   628→              child: NotificationListener<ScrollNotification>(
   629→                onNotification: (scrollInfo) {
   630→                  // Task 38 Req 5 — trigger at 80% scroll instead of 100%
   631→                  // so the next page kicks in BEFORE the user hits the
   632→                  // bottom (smoother infinite scroll). Combined with the
   633→                  // post-frame auto-load below, this also handles the case
   634→                  // where the first page didn't even fill the viewport
   635→                  // (e.g., on a large tablet).
   636→                  final trigger = scrollInfo.metrics.pixels >=
   637→                      scrollInfo.metrics.maxScrollExtent * 0.8;
   638→                  if (trigger && !_isLoadingMore && _hasMore) {
   639→                    _loadMore();
   640→                  }
   641→                  return false;
   642→                },
   643→                child: _filteredMovies.isEmpty
   644→                    ? Center(
   645→                        child: Column(
   646→                          mainAxisAlignment: MainAxisAlignment.center,
   647→                          children: [
   648→                            Icon(
   649→                              Icons.movie_filter_outlined,
   650→                              size: 64,
   651→                              color: theme.colorScheme.onSurface.withOpacity(0.3),
   652→                            ),
   653→                            const SizedBox(height: 16),
   654→                            Text(
   655→                              'No ${widget.title} yet',
   656→                              style: theme.textTheme.bodyLarge?.copyWith(
   657→                                color: theme.colorScheme.onSurface.withOpacity(0.5),
   658→                              ),
   659→                            ),
   660→                          ],
   661→                        ),
   662→                      )
   663→                    : GridView.builder(
   664→                        padding: const EdgeInsets.symmetric(horizontal: 6, vertical: 6),
   665→                        gridDelegate: const SliverGridDelegateWithFixedCrossAxisCount(
   666→                          crossAxisCount: 3,
   667→                          childAspectRatio: 0.53,
   668→                          crossAxisSpacing: 6,
   669→                          mainAxisSpacing: 6,
   670→                        ),
   671→                        itemCount: _filteredMovies.length + (_isLoadingMore ? 6 : 0),
   672→                        itemBuilder: (context, index) {
   673→                          if (index >= _filteredMovies.length) {
   674→                            return const Center(child: CircularProgressIndicator());
   675→                          }
   676→                          final movie = _filteredMovies[index];
   677→                          return MovieCard(
   678→                            movie: movie,
   679→                            onTap: () {
   680→                              Navigator.push(
   681→                                context,
   682→                                MaterialPageRoute(
   683→                                  builder: (_) => movie.type == 'series'
   684→                                      ? SeriesDetailScreen(slug: movie.slug)
   685→                                      : MovieDetailScreen(slug: movie.slug),
   686→                                ),
   687→                              );
   688→                            },
   689→                          );
   690→                        },
   691→                      ),
   692→              ),
   693→            ),
   694→    );
   695→  }
