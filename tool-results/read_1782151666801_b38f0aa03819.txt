    36→class _HomePageState extends State<HomePage> {
    37→  int _currentIndex = kHomeTab;
    38→
    39→  // Global keys for tab pages to allow refresh
    40→  final GlobalKey<State<MoviesPage>> _moviesKey = GlobalKey();
    41→  final GlobalKey<State<SeriesPage>> _seriesKey = GlobalKey();
    42→  // Home has its own key so we can call its public refresh() method after
    43→  // returning from content-modifying screens (TMDB Generator, Admin Panel,
    44→  // Add Movie, Edit Movie, Batch Import). Without this, Home keeps showing
    45→  // stale data because IndexedStack keeps HomeScreen alive across tab
    46→  // switches and does NOT auto-reload on Navigator.pop().
    47→  final GlobalKey<State<HomeScreen>> _homeKey = GlobalKey();
    48→
    49→  // Tab pages - created once to preserve state across tab switches
    50→  late final List<Widget> _bottomNavPages;
    51→
    52→  // Double-back-to-exit: prevent accidental app exit from root route
    53→  bool _canExit = false;
    54→  Timer? _exitTimer;
    55→
    56→  // Real app version (fetched from package_info_plus)
    57→  String _appVersion = '';
    58→
    59→  @override
    60→  void initState() {
    61→    super.initState();
    62→    _loadAppVersion();
    63→    _bottomNavPages = [
    64→      HomeScreen(
    65→        key: _homeKey,
    66→        onNavigateToTab: (index) {
    67→          // Switch tab first
    68→          setState(() => _currentIndex = index);
    69→          // Trigger data refresh with skeleton loading when navigating via "More" button
    70→          // This ensures skeleton shows first, then fresh data loads
    71→          if (index == kMoviesTab) {
    72→            // Use addPostFrameCallback to ensure the tab is visible before refreshing
    73→            WidgetsBinding.instance.addPostFrameCallback((_) {
    74→              final state = _moviesKey.currentState;
    75→              if (state != null) (state as dynamic).onTabSelected();
    76→            });
    77→          } else if (index == kSeriesTab) {
    78→            WidgetsBinding.instance.addPostFrameCallback((_) {
    79→              final state = _seriesKey.currentState;
    80→              if (state != null) (state as dynamic).onTabSelected();
    81→            });
    82→          }
    83→        },
    84→      ),
    85→      MoviesPage(key: _moviesKey),
    86→      SeriesPage(key: _seriesKey),
    87→      const SettingsPage(),
    88→    ];
    89→  }

    91→  @override
    92→  void dispose() {
    93→    _exitTimer?.cancel();
    94→    super.dispose();
    95→  }

    97→  // Fetch the real app version (e.g. "1.9.0+15") from pubspec at runtime
    98→  Future<void> _loadAppVersion() async {
    99→    try {
   100→      final info = await PackageInfo.fromPlatform();
   101→      final version = info.version;
   102→      final build = info.buildNumber;
   103→      final formatted = build.isEmpty ? version : '$version+$build';
   104→      if (mounted) {
   105→        setState(() => _appVersion = formatted);
   106→      }
   107→    } catch (_) {
   108→      // Fallback: leave empty so UI can fall back to a static label
   109→    }
   110→  }

   112→  /// Silently refresh the Home tab's data. Called after returning from
   113→  /// content-modifying screens (TMDB Generator, Admin Panel, etc.) so that
   114→  /// newly-added or edited posts appear at the top of the Home feed
   115→  /// without requiring the user to pull-to-refresh.
   116→  ///
   117→  /// Safe to call when Home is not the current tab — IndexedStack keeps
   118→  /// HomeScreen alive, so the GlobalKey state is always available.
   119→  void _refreshHomeIfMounted() {
   120→    final state = _homeKey.currentState;
   121→    if (state != null) {
   122→      // ignore: avoid_dynamic_calls — HomeScreen.refresh() is a public API
   123→      (state as dynamic).refresh();
   124→    }
   125→  }

   241→  Widget _buildBottomNav(AppConfig appConfig, ThemeData theme) {
   242→    final isDark = theme.brightness == Brightness.dark;
   243→    final unselectedColor = isDark ? Colors.grey.shade500 : Colors.grey.shade600;
   244→    const selectedColor = Color(0xFFE50914);
   245→
   246→    return Container(
   247→      decoration: BoxDecoration(
   248→        color: isDark ? const Color(0xFF121212) : Colors.white,
   249→        boxShadow: [
   250→          BoxShadow(
   251→            color: isDark ? Colors.black26 : Colors.grey.shade300,
   252→            blurRadius: 4,
   253→            spreadRadius: 0,
   254→            offset: const Offset(0, -1),
   255→          ),
   256→        ],
   257→      ),
   258→      child: NavigationBarTheme(
   259→        data: NavigationBarThemeData(
   260→          backgroundColor: isDark ? const Color(0xFF121212) : Colors.white,
   261→          indicatorColor: Colors.transparent,
   262→          height: 64,
   263→          labelBehavior: NavigationDestinationLabelBehavior.alwaysShow,
   264→          surfaceTintColor: Colors.transparent,
   265→          overlayColor: WidgetStateProperty.all(Colors.transparent),
   266→          iconTheme: WidgetStateProperty.resolveWith<IconThemeData?>((states) {
   267→            final isSelected = states.contains(WidgetState.selected);
   268→            return IconThemeData(
   269→              size: 24,
   270→              color: isSelected ? selectedColor : unselectedColor,
   271→            );
   272→          }),
   273→          labelTextStyle: WidgetStateProperty.resolveWith<TextStyle?>((states) {
   274→            final isSelected = states.contains(WidgetState.selected);
   275→            return TextStyle(
   276→              fontSize: 11,
   277→              fontWeight: isSelected ? FontWeight.w600 : FontWeight.normal,
   278→              color: isSelected ? selectedColor : unselectedColor,
   279→            );
   280→          }),
   281→        ),
   282→        child: NavigationBar(
   283→        selectedIndex: _currentIndex,
   284→        onDestinationSelected: (index) {
   285→          // If user taps Home tab while already on Home, treat it as a
   286→          // refresh trigger (silently — no skeleton). This mirrors the
   287→          // common "tap the active tab to refresh" pattern in apps like
   288→          // Twitter / Instagram. Otherwise just switch tabs.
   289→          if (index == _currentIndex && index == kHomeTab) {
   290→            _refreshHomeIfMounted();
   291→            return;
   292→          }
   293→          setState(() {
   294→            _currentIndex = index;
   295→          });
   296→          // Trigger data refresh when switching to Movies/Series tabs
   297→          if (index == kMoviesTab) {
   298→            final state = _moviesKey.currentState;
   299→            if (state != null) (state as dynamic).onTabSelected();
   300→          } else if (index == kSeriesTab) {
   301→            final state = _seriesKey.currentState;
   302→            if (state != null) (state as dynamic).onTabSelected();
   303→          } else if (index == kHomeTab) {
   304→            // Returning to Home — silently refresh in case the user
   305→            // added/edited content elsewhere since Home was last visible.
   306→            _refreshHomeIfMounted();
   307→          }
   308→        },
   309→        destinations: [
   310→          NavigationDestination(
   311→            icon: const Icon(Icons.home_outlined),
   312→            selectedIcon: const Icon(Icons.home),
   313→            label: appConfig.translate('home'),
   314→          ),
   315→          NavigationDestination(
   316→            icon: const Icon(Icons.movie_outlined),
   317→            selectedIcon: const Icon(Icons.movie),
   318→            label: appConfig.translate('movies'),
   319→          ),
   320→          NavigationDestination(
   321→            icon: const Icon(Icons.tv_outlined),
   322→            selectedIcon: const Icon(Icons.tv),
   323→            label: appConfig.translate('series'),
   324→          ),
   325→          NavigationDestination(
   326→            icon: const Icon(Icons.settings_outlined),
   327→            selectedIcon: const Icon(Icons.settings),
   328→            label: appConfig.translate('settings'),
   329→          ),
   330→        ],
   331→      ),
   332→      ),
   333→    );
   334→  }

   622→  /// VIP status card shown inside the drawer header.

   624→  /// - Non-VIP users → outlined grey/orange card with "VIP Not Active" + "Tap to upgrade"
   625→  ///   (tapping opens the VipPage where they can purchase VIP via Telegram)
   626→  Widget _buildVipStatusCard(AppConfig appConfig, bool isDark) {
   627→    final isVip = appConfig.isCurrentUserVip;
   628→    final isAdmin = appConfig.isCurrentUserAdmin;
   629→    // Admins implicitly have all VIP perks — show VIP-style badge for them too
   630→    final showAsVip = isVip || isAdmin;
   631→
   632→    if (showAsVip) {
   633→      // Format the VIP expiry date for display (if available)
   634→      String expiryDisplay = '';
   635→      final rawExpiry = appConfig.currentUser?['vipExpiry'] as String?;
   636→      if (rawExpiry != null && rawExpiry.isNotEmpty) {
   637→        final dt = DateTime.tryParse(rawExpiry);
   638→        if (dt != null) {
   639→          expiryDisplay =
   640→              '${dt.year}-${dt.month.toString().padLeft(2, '0')}-${dt.day.toString().padLeft(2, '0')}';
   641→        }
   642→      }
   643→
   644→      return Container(
   645→        width: double.infinity,
   646→        padding: const EdgeInsets.symmetric(horizontal: 14, vertical: 12),
   647→        decoration: BoxDecoration(
   648→          borderRadius: BorderRadius.circular(12),
   649→          gradient: const LinearGradient(
   650→            colors: [Color(0xFFE50914), Color(0xFFB20710)],
   651→            begin: Alignment.topLeft,
   652→            end: Alignment.bottomRight,
   653→          ),
   654→          boxShadow: [
   655→            BoxShadow(
   656→              color: const Color(0xFFE50914).withOpacity(0.35),
   657→              blurRadius: 10,
   658→              offset: const Offset(0, 3),
   659→            ),
   660→          ],
   661→        ),
   662→        child: Row(
   663→          children: [
   664→            const Icon(
   665→              Icons.workspace_premium_rounded,
   666→              color: Colors.white,
   667→              size: 24,
   668→            ),
   669→            const SizedBox(width: 10),
   670→            Expanded(
   671→              child: Column(
   672→                crossAxisAlignment: CrossAxisAlignment.start,
   673→                mainAxisSize: MainAxisSize.min,
   674→                children: [
   675→                  Text(
   676→                    isAdmin
   677→                        ? '${appConfig.translate('vip_active')} · Admin'
   678→                        : appConfig.translate('vip_active'),
   679→                    style: const TextStyle(
   680→                      color: Colors.white,
   681→                      fontSize: 13,
   682→                      fontWeight: FontWeight.bold,
   683→                    ),
   684→                  ),
   685→                  const SizedBox(height: 2),
   686→                  Text(
   687→                    expiryDisplay.isEmpty
   688→                        ? appConfig.translate('vip_active_desc')
   689→                        : appConfig
   690→                            .translate('vip_expires_on')
   691→                            .replaceAll('{date}', expiryDisplay),
   692→                    style: TextStyle(
   693→                      color: Colors.white.withOpacity(0.85),
   694→                      fontSize: 11,
   695→                    ),
   696→                  ),
   697→                ],
   698→              ),
   699→            ),
   700→          ],
   701→        ),
   702→      );
   703→    }
   704→
   705→    // Non-VIP card — tap to open VipPage and purchase VIP
   706→    return Material(
   707→      color: Colors.transparent,
   708→      child: InkWell(
   709→        borderRadius: BorderRadius.circular(12),
   710→        onTap: () {
   711→          Navigator.pop(context); // close drawer
   712→          Navigator.push(
   713→            context,
   714→            MaterialPageRoute(builder: (_) => const VipPage()),
   715→          );
   716→        },
   717→        child: Container(
   718→          width: double.infinity,
   719→          padding: const EdgeInsets.symmetric(horizontal: 14, vertical: 12),
   720→          decoration: BoxDecoration(
   721→            borderRadius: BorderRadius.circular(12),
   722→            color: isDark ? Colors.white.withOpacity(0.05) : Colors.black.withOpacity(0.04),
   723→            border: Border.all(
   724→              color: const Color(0xFFE50914).withOpacity(0.5),
   725→              width: 1.2,
   726→            ),
   727→          ),
   728→          child: Row(
   729→            children: [
   730→              Icon(
   731→                Icons.lock_outline,
   732→                color: const Color(0xFFE50914).withOpacity(0.9),
   733→                size: 24,
   734→              ),
   735→              const SizedBox(width: 10),
   736→              Expanded(
   737→                child: Column(
   738→                  crossAxisAlignment: CrossAxisAlignment.start,
   739→                  mainAxisSize: MainAxisSize.min,
   740→                  children: [
   741→                    Text(
   742→                      appConfig.translate('vip_inactive'),
   743→                      style: TextStyle(
   744→                        color: isDark ? Colors.white : Colors.black87,
   745→                        fontSize: 13,
   746→                        fontWeight: FontWeight.bold,
   747→                      ),
   748→                    ),
   749→                    const SizedBox(height: 2),
   750→                    Text(
   751→                      appConfig.translate('vip_inactive_desc'),
   752→                      style: TextStyle(
   753→                        color: isDark ? Colors.white60 : Colors.black54,
   754→                        fontSize: 11,
   755→                      ),
   756→                    ),
   757→                  ],
   758→                ),
   759→              ),
   760→              const SizedBox(width: 6),
   761→              Container(
   762→                padding: const EdgeInsets.symmetric(horizontal: 8, vertical: 4),
   763→                decoration: BoxDecoration(
   764→                  color: const Color(0xFFE50914),
   765→                  borderRadius: BorderRadius.circular(6),
   766→                ),
   767→                child: Text(
   768→                  appConfig.translate('vip_get_now'),
   769→                  style: const TextStyle(
   770→                    color: Colors.white,
   771→                    fontSize: 10,
   772→                    fontWeight: FontWeight.bold,
   773→                  ),
   774→                ),
   775→              ),
   776→            ],
   777→          ),
   778→        ),

   780→    );
