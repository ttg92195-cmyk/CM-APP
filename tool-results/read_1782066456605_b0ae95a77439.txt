    43→  /// Human-readable file size, e.g. "12.4 KB" or "1.8 MB".
    44→  String get sizeFormatted {
    45→    const units = ['B', 'KB', 'MB', 'GB'];
    46→    var size = sizeBytes.toDouble();
    47→    var unit = 0;
    48→    while (size >= 1024 && unit < units.length - 1) {
    49→      size /= 1024;
    50→      unit++;
    51→    }

   404→/// Service that orchestrates a Batch Import flow:
   405→///   1. parseFile()    — read JSON file, deserialize each entry, capture parse errors
   406→///   2. classify()     — for each valid item, check Firestore for tmdbId/slug duplicates
   407→///   3. runImport()    — sequentially call FirestoreContentService.addMovie() with progress
   408→///
   409→/// The service deliberately does NOT use WriteBatch — it reuses the existing
   410→/// addMovie() function which already handles duplicate detection, counter sync,
   411→/// and idempotent updates. This keeps the import logic safe and consistent
   412→/// with the rest of the app.
   413→class BatchImportService {
   414→  final FirestoreContentService _contentService;
   415→  final FirebaseFirestore _firestore;
   416→
   417→  BatchImportService({
   418→    FirestoreContentService? contentService,
   419→    FirebaseFirestore? firestore,
   420→  })  : _contentService = contentService ?? FirestoreContentService(),
   421→        _firestore = firestore ?? FirebaseFirestore.instance;
   422→
   423→  // ===========================================================================
   424→  // PHASE 3d — MEMORY SAFETY GUARDS
   425→  // ===========================================================================
   426→  //
   427→  // Hard caps to protect low-end devices from OOM-kills and Firestore from
   428→  // billing surprises. Tuned conservatively — typical admin exports are well
   429→  // under both limits.
   430→  //
   431→  //   maxFileSizeBytes:   a 50 MB JSON file decodes to ~300 MB of heap on a
   432→  //                       mid-range phone, which is enough to trigger Android's
   433→  //                       low-memory killer. Admins with bigger datasets should
   434→  //                       split the file or use Firestore's native import tools.
   435→  //
   436→  //   maxItemsPerImport:  5000 items × ~2 Firestore reads each (classify) +
   437→  //                       1 write each (addMovie) = ~15000 Firestore ops.
   438→  //                       That's still within free-tier daily quotas but takes
   439→  //                       ~10-15 minutes to import on a stable connection.
   440→  //                       Beyond this, the linear progress UX feels broken.
   441→
   442→  /// Absolute upper bound on the JSON file size we'll attempt to parse.
   443→  /// Files larger than this are refused outright — the admin should split
   444→  /// them or use Firestore's native import tools.
   445→  static const int maxFileSizeBytes = 50 * 1024 * 1024; // 50 MB
   446→
   447→  /// Absolute upper bound on the number of items we'll process in one import.
   448→  /// Larger arrays are refused to avoid (a) hours-long imports with no
   449→  /// resumability, (b) hitting Firestore daily write quotas accidentally,
   450→  /// (c) overwhelming the audit-log doc size when many items fail.
   451→  static const int maxItemsPerImport = 5000;
   452→
   453→  /// Soft warning threshold (5 MB). The UI uses this to show a confirmation
   454→  /// dialog before kicking off parse — gives the admin a chance to back out
   455→  /// if they accidentally picked a 40 MB file.
   456→  static const int largeFileWarningBytes = 5 * 1024 * 1024; // 5 MB
   457→
   458→  /// Human-readable file size, e.g. "12.4 KB" or "1.8 MB".
   459→  /// Kept as a static helper so the UI can format the same way the service
   460→  /// reports sizes in error messages.
   461→  static String formatFileSize(int bytes) {
   462→    const units = ['B', 'KB', 'MB', 'GB'];
   463→    var size = bytes.toDouble();
   464→    var unit = 0;
   465→    while (size >= 1024 && unit < units.length - 1) {
   466→      size /= 1024;
   467→      unit++;
   468→    }
   469→    return '${size.toStringAsFixed(size < 10 ? 1 : 0)} ${units[unit]}';
   470→  }

   473→  // PHASE 0 — EXPORT (BACKUP)
   474→  // ===========================================================================

   476→  /// Export every movie in the Firestore `movies` collection to a JSON file
   477→  /// on the local device. The file is written to [outputDir] (or the app's
   478→  /// documents directory if null) with a timestamped file name like
   479→  /// `cm_movies_backup_2026-06-19_14-30-00.json`.
   480→  ///
   481→  /// This is a "safety net" feature: the admin should run it before any
   482→  /// large Batch Import, so if something goes wrong they have an exact
   483→  /// snapshot of the database to restore from (by importing the backup file
   484→  /// back through BatchImportPage).
   485→  ///
   486→  /// Pagination: fetches in pages of [pageSize] (default 200) to keep memory
   487→  /// usage flat regardless of collection size. A 1000-movie export uses
   488→  /// ~40 MB heap during the write phase.
   489→  ///
   490→  /// The exported JSON is an array of raw Firestore document maps, wrapped
   491→  /// in a top-level object with metadata:
   492→  ///   {
   493→  ///     "_meta": {
   494→  ///       "exportedAt": "2026-06-19T14:30:00",
   495→  ///       "count": 1234,
   496→  ///       "source": "cm_movies"
   497→  ///     },
   498→  ///     "movies": [ { ...full movie doc... }, ... ]
   499→  ///   }
   500→  ///
   501→  /// The "movies" wrapper key is recognised by parseJsonString(), so the
   502→  /// exported file can be re-imported as-is to restore.
   503→  Future<BatchExportResult> exportAllMovies({
   504→    String? outputDir,
   505→    int pageSize = 200,
   506→    void Function(BatchExportProgress)? onProgress,
   507→  }) async {
   508→    final moviesRef = _firestore.collection('movies');
   509→    final allDocs = <Map<String, dynamic>>[];
   510→    DocumentSnapshot? lastDoc;
   511→    bool hasMore = true;
   512→
   513→    // Paginated fetch — Firestore .get() returns at most the page size, and
   514→    // we follow with startAfterDocument() until a page comes back short.
   515→    while (hasMore) {
   516→      Query query = moviesRef
   517→          .orderBy('createdAt', descending: true)
   518→          .limit(pageSize);
   519→
   520→      if (lastDoc != null) {
   521→        query = query.startAfterDocument(lastDoc);
   522→      }
   523→
   524→      final snapshot = await query.get();
   525→
   526→      if (snapshot.docs.isEmpty) {
   527→        hasMore = false;
   528→        break;
   529→      }
   530→
   531→      for (final doc in snapshot.docs) {
   532→        final data = doc.data() as Map<String, dynamic>;
   533→        // Strip server-only fields that wouldn't round-trip cleanly.
   534→        // createdAt/updatedAt are Timestamps on the server; we keep them as
   535→        // ISO strings so the JSON is human-readable and parseable later.
   536→        final cleaned = _serializeMovieDoc(data, doc.id);
   537→        allDocs.add(cleaned);
   538→      }
   539→
   540→      lastDoc = snapshot.docs.last;
   541→      hasMore = snapshot.docs.length >= pageSize;
   542→
   543→      onProgress?.call(BatchExportProgress(
   544→        exportedSoFar: allDocs.length,
   545→        hasMore: hasMore,
   546→      ));
   547→    }
   548→
   549→    // Build the output JSON. We do NOT use jsonEncode with the entire list
   550→    // in one shot for very large datasets — but in practice 1000 movies
   551→    // produces ~5 MB of JSON which encodes in <500 ms and uses ~25 MB heap.
   552→    // For 10,000+ movies we'd switch to a streaming encoder, but that's a
   553→    // Phase 4+ concern.
   554→    final now = DateTime.now();
   555→    final payload = <String, dynamic>{
   556→      '_meta': {
   557→        'exportedAt': now.toIso8601String(),
   558→        'count': allDocs.length,
   559→        'source': 'cm_movies',
   560→        'appVersion': 'batch_import_v1',
   561→      },
   562→      'movies': allDocs,
   563→    };
   564→
   565→    final jsonStr = const JsonEncoder.withIndent('  ').convert(payload);
   566→    final bytes = utf8.encode(jsonStr);
   567→
   568→    // Resolve output directory.
   569→    Directory dir;
   570→    if (outputDir != null && outputDir.isNotEmpty) {
   571→      dir = Directory(outputDir);
   572→      if (!await dir.exists()) {
   573→        await dir.create(recursive: true);
   574→      }
   575→    } else {
   576→      // Default to the app's documents directory — always writable, no
   577→      // storage permission needed on Android 10+.
   578→      dir = await _defaultExportDir();
   579→    }
   580→
   581→    final fileName = _formatBackupFileName(now);
   582→    final file = File('${dir.path}/$fileName');
   583→    await file.writeAsBytes(bytes, flush: true);
   584→
   585→    return BatchExportResult(
   586→      filePath: file.path,
   587→      count: allDocs.length,
   588→      sizeBytes: bytes.length,
   589→      exportedAt: now,
   590→    );
   591→  }
   592→
   593→  /// Default export directory: app documents dir + '/batch_import_backups'.
   594→  /// Created on first call. We use a subdirectory so backups don't pollute
   595→  /// the root documents directory.
   596→  Future<Directory> _defaultExportDir() async {
   597→    // Use the app's documents directory — always writable, no storage
   598→    // permission needed on Android 10+. On Android, this resolves to
   599→    // /data/data/<package>/app_flutter (or similar).
   600→    final baseDir = await getApplicationDocumentsDirectory();
   601→    final exportDir = Directory('${baseDir.path}/batch_import_backups');
   602→    if (!await exportDir.exists()) {
   603→      await exportDir.create(recursive: true);
   604→    }
   605→    return exportDir;
   606→  }

   608→  /// Convert a raw Firestore movie document into a JSON-serializable map.
   609→  /// Firestore Timestamp fields become ISO-8601 strings; everything else
   610→  /// passes through unchanged. The document ID is preserved as 'id' so
   611→  /// re-import can detect duplicates by tmdbId/slug (still safe).
   612→  Map<String, dynamic> _serializeMovieDoc(
   613→    Map<String, dynamic> data,
   614→    String docId,
   615→  ) {
   616→    final out = <String, dynamic>{};
   617→
   618→    data.forEach((key, value) {
   619→      out[key] = _serializeValue(value);
   620→    });
   621→
   622→    // Always preserve the Firestore document ID for traceability.
   623→    // addMovie() doesn't use this field (it uses tmdbId/slug), so it's
   624→    // safe to include — and useful for manual diffing if a restore goes wrong.
   625→    if (!out.containsKey('id')) {
   626→      out['id'] = docId;
   627→    }
   628→
   629→    return out;
   630→  }
   631→
   632→  /// Recursively serialize a Firestore value into JSON-safe form.
   633→  /// Timestamps → ISO string. GeoPoint → {lat, lng}. DocumentReference → path.
   634→  dynamic _serializeValue(dynamic value) {
   635→    if (value == null) return null;
   636→    if (value is Timestamp) return value.toDate().toIso8601String();
   637→    if (value is DateTime) return value.toIso8601String();
   638→    if (value is Map) {
   639→      final out = <String, dynamic>{};
   640→      value.forEach((k, v) => out[k.toString()] = _serializeValue(v));
   641→      return out;
   642→    }
   643→    if (value is List) {
   644→      return value.map(_serializeValue).toList();
   645→    }
   646→    // Primitives (String, int, double, bool) pass through.
   647→    return value;
   648→  }

   650→  /// Build a safe, filesystem-friendly file name for the backup.
   651→  /// Format: cm_movies_backup_YYYY-MM-DD_HH-MM-ss.json
   652→  String _formatBackupFileName(DateTime now) {
   653→    String two(int v) => v.toString().padLeft(2, '0');
   654→    return 'cm_movies_backup_'
   655→        '${now.year}-${two(now.month)}-${two(now.day)}_'
   656→        '${two(now.hour)}-${two(now.minute)}-${two(now.second)}'
   657→        '.json';
   658→  }

   664→  /// Read the JSON file from [path] and parse it into [BatchImportItem]s.
   665→  ///
   666→  /// Accepted JSON shapes:
   667→  ///   A) A JSON array of movie objects:   [ { ... }, { ... }, ... ]
   668→  ///   B) An object with a 'movies' key:   { "movies": [ ... ] }
   669→  ///   C) An object with a 'data' key:     { "data":   [ ... ] }
   670→  ///
   671→  /// Each item is validated against a minimal schema (title required, type
   672→  /// must be 'movie' or 'series' if present). Invalid items are kept in the
   673→  /// returned list but flagged with status == invalid so the UI can preview
   674→  /// them; they will be skipped during the import phase.
   675→  ///
   676→  /// Memory safety: files larger than [maxFileSizeBytes] are refused with a
   677→  /// clear [BatchImportException] before any bytes are read into memory.
   678→  /// The UI is expected to ALSO warn the admin before calling this method
   679→  /// when the file is above [largeFileWarningBytes] (5 MB) — that's a
   680→  /// soft UX nudge, this is a hard guard.
   681→  Future<BatchParseResult> parseFile(String path) async {
   682→    final file = File(path);
   683→    if (!await file.exists()) {
   684→      throw BatchImportException('File not found: $path');
   685→    }
   686→
   687→    final fileSize = await file.length();
   688→    if (fileSize == 0) {
   689→      return const BatchParseResult(
   690→        items: [],
   691→        parseErrors: [],
   692→        isEmpty: true,
   693→      );
   694→    }
   695→
   696→    // Hard cap: refuse to even read files above the absolute max size.
   697→    // This prevents OOM-kills on low-end devices where the OS would
   698→    // otherwise let us allocate ~300 MB of heap and then crash the app.
   699→    if (fileSize > maxFileSizeBytes) {
   700→      throw BatchImportException(
   701→        'File is too large to import safely.\n'
   702→        'Size: ${formatFileSize(fileSize)} (limit: '
   703→        '${formatFileSize(maxFileSizeBytes)}).\n'
   704→        'Please split the file into smaller chunks of ~1000 movies each, '
   705→        'or use Firestore\'s native import tools for very large datasets.',
   706→      );
   707→    }
   708→
   709→    // Read the entire file. For very large files (>20 MB) we would need a
   710→    // streaming parser, but the typical admin export is well under 5 MB.
   711→    // A 5 MB JSON file with ~1000 movies parses in <1 second on a mid-range
   712→    // phone and uses ~30 MB of heap — well within Flutter's default budget.
   713→    final raw = await file.readAsString();
   714→    return parseJsonString(raw);
   715→  }
   716→
   717→  /// Same as [parseFile] but operates on an in-memory JSON string.
   718→  /// Useful for tests and for parsing JSON pasted into a text field.
   719→  BatchParseResult parseJsonString(String raw) {
   720→    if (raw.trim().isEmpty) {
   721→      return const BatchParseResult(
   722→        items: [],
   723→        parseErrors: [],
   724→        isEmpty: true,
   725→      );
   726→    }
   727→
   728→    dynamic decoded;
   729→    try {
   730→      decoded = jsonDecode(raw);
   731→    } catch (e) {
   732→      throw BatchImportException(
   733→        'Invalid JSON: ${e.toString()}',
   734→      );
   735→    }
   736→
   737→    List<dynamic>? array;
   738→    if (decoded is List) {
   739→      array = decoded;
   740→    } else if (decoded is Map) {
   741→      // Accept either 'movies' or 'data' as the wrapper key.
   742→      final inner = decoded['movies'] ?? decoded['data'];
   743→      if (inner is List) {
   744→        array = inner;
   745→      } else {
   746→        throw BatchImportException(
   747→          'Expected a JSON array, or an object with a "movies" or "data" key '
   748→          'containing an array.',
   749→        );
   750→      }
   751→    } else {
   752→      throw BatchImportException(
   753→        'Expected a JSON array or object at the top level.',
   754→      );
   755→    }
   756→
   757→    if (array.isEmpty) {
   758→      return const BatchParseResult(
   759→        items: [],
   760→        parseErrors: [],
   761→        isEmpty: true,
   762→      );
   763→    }
   764→
   765→    // Hard cap on item count. Beyond this the import becomes impractical:
   766→    //   - Linear Firestore reads/writes would take 10+ minutes
   767→    //   - The audit-log doc would be oversized if many items failed
   768→    //   - Memory pressure from holding all BatchImportItem objects in RAM
   769→    //   - No resumability — if the app dies mid-import, all progress is lost
   770→    if (array.length > maxItemsPerImport) {
   771→      throw BatchImportException(
   772→        'Too many items in the JSON array.\n'
   773→        'Count: ${array.length} (limit: $maxItemsPerImport).\n'
   774→        'Please split the file into smaller batches of ~1000 movies each. '
   775→        'You can re-run Batch Import as many times as you want — duplicates '
   776→        'are detected and skipped automatically.',
   777→      );
   778→    }
   779→
   780→    final items = <BatchImportItem>[];
   781→    final parseErrors = <Map<String, dynamic>>[];
   782→
   783→    for (var i = 0; i < array.length; i++) {
   784→      final entry = array[i];
   785→      if (entry is! Map) {
   786→        parseErrors.add({
   787→          'index': i + 1,
   788→          'error': 'Entry is not a JSON object (got ${entry.runtimeType}).',
   789→        });
   790→        continue;
   791→      }
   792→
   793→      // Make a mutable, deep-cast copy so addMovie() can mutate it safely.
   794→      final data = _deepCast(entry);
   795→      final validationError = _validate(data);
   796→
   797→      items.add(BatchImportItem(
   798→        sourceIndex: i + 1,
   799→        data: data,
   800→        status: validationError == null
   801→            ? BatchItemStatus.willCreate // tentative — classify() will refine
   802→            : BatchItemStatus.invalid,
   803→        validationError: validationError,
   804→      ));
   805→    }
   806→
   807→    return BatchParseResult(
   808→      items: items,
   809→      parseErrors: parseErrors,
   810→      isEmpty: false,
   811→    );
   812→  }

   814→  // ===========================================================================
   815→  // PHASE 2 — CLASSIFY (preview)
   816→  // ===========================================================================
   817→
   818→  /// For each non-invalid item, look up Firestore by tmdbId then slug.
   819→  /// Sets item.status to willCreate or willUpdate accordingly.
   820→  ///
   821→  /// This is the slowest step because it issues 1-2 Firestore reads per item.
   822→  /// For 1000 items this is ~2-3 seconds of network round-trips.
   823→  Future<void> classify(List<BatchImportItem> items) async {
   824→    for (final item in items) {
   825→      if (item.status == BatchItemStatus.invalid) continue;
   826→
   827→      bool exists = false;
   828→
   829→      // PRIORITY 1: tmdbId (most reliable)
   830→      final tmdbId = item.data['tmdbId'];
   831→      if (!exists && tmdbId != null) {
   832→        try {
   833→          final existing = await _contentService.findByTmdbId(tmdbId);
   834→          if (existing != null) exists = true;
   835→        } catch (e) {
   836→          debugPrint('classify: findByTmdbId failed for $tmdbId: $e');
   837→          // Treat as not-found — addMovie() will re-check and handle safely.
   838→        }
   839→      }
   840→
   841→      // PRIORITY 2: slug (auto-generated if missing)
   842→      if (!exists) {
   843→        final slug = (item.data['slug']?.toString().isNotEmpty ?? false)
   844→            ? item.data['slug'].toString()
   845→            : _generateSlug(item.data['title']?.toString() ?? '');
   846→        if (slug.isNotEmpty) {
   847→          try {
   848→            final snap = await _firestore
   849→                .collection('movies')
   850→                .where('slug', isEqualTo: slug)
   851→                .limit(1)
   852→                .get();
   853→            if (snap.docs.isNotEmpty) exists = true;
   854→          } catch (e) {
   855→            debugPrint('classify: slug lookup failed for "$slug": $e');
   856→          }
   857→        }
   858→      }
   859→
   860→      item.status =
   861→          exists ? BatchItemStatus.willUpdate : BatchItemStatus.willCreate;
   862→    }
   863→  }
   864→
   865→  /// Quick aggregated counts — call after [classify].
   866→  BatchImportCounts tally(List<BatchImportItem> items) {
   867→    int willCreate = 0, willUpdate = 0, invalid = 0;
   868→    for (final item in items) {
   869→      switch (item.status) {
   870→        case BatchItemStatus.willCreate:
   871→          willCreate++;
   872→          break;
   873→        case BatchItemStatus.willUpdate:
   874→          willUpdate++;
   875→          break;
   876→        case BatchItemStatus.invalid:
   877→          invalid++;
   878→          break;
   879→      }
   880→    }
   881→    return BatchImportCounts(
   882→      total: items.length,
   883→      willCreate: willCreate,
   884→      willUpdate: willUpdate,
   885→      invalid: invalid,
   886→    );
   887→  }

   890→  // PHASE 3 — IMPORT
   891→  // ===========================================================================
   892→
   893→  /// Sequentially import all actionable items (willCreate + willUpdate).
   894→  /// Skips invalid items.
   895→  ///
   896→  /// Calls [onProgress] after every item so the UI can update its progress bar.
   897→  /// Stops early if [shouldStop] returns true — useful for a Cancel button.
   898→  ///
   899→  /// If [auditContext] is provided, an entry is written to the
   900→  /// `batch_imports` Firestore collection after the import finishes
   901→  /// (whether it succeeded, partially failed, or was cancelled). The audit
   902→  /// write is wrapped in a try/catch so a Firestore permission or network
   903→  /// error never causes the import itself to appear failed.
   904→  ///
   905→  /// Returns a [BatchImportResult] with per-item outcomes.
   906→  Future<BatchImportResult> runImport(
   907→    List<BatchImportItem> items, {
   908→    void Function(BatchImportProgress)? onProgress,
   909→    bool Function()? shouldStop,
   910→    BatchImportAuditContext? auditContext,
   911→  }) async {
   912→    final startedAt = DateTime.now();
   913→
   914→    // =========================================================================
   915→    // AUDIT C2 — verify admin ONCE here, then pass skipAdminCheck:true to
   916→    // addMovie() for every item. Before this fix, addMovie() called
   917→    // _requireAdmin() on every invocation, which issued a Firestore read to
   918→    // the user doc per item. For a 1000-movie import that was 1000 wasted
   919→    // reads just to re-confirm what we already knew at the start of the run.
   920→    //
   921→    // Defense-in-depth is preserved: verifyAdmin() throws if the user is
   922→    // not an admin, and runImport() aborts before touching Firestore.
   923→    // Firestore security rules still apply to every individual write, so
   924→    // a privilege escalation (admin demoted mid-import) is contained.
   925→    // =========================================================================
   926→    await _contentService.verifyAdmin();
   927→
   928→    final actionable = items
   929→        .where((i) => i.status != BatchItemStatus.invalid)
   930→        .toList(growable: false);
   931→
   932→    int created = 0;
   933→    int updated = 0;
   934→    int failed = 0;
   935→    int skipped = items.length - actionable.length;
   936→    bool cancelled = false;
   937→
   938→    final total = actionable.length;
   939→    for (var i = 0; i < total; i++) {
   940→      if (shouldStop != null && shouldStop()) {
   941→        // Mark remaining items as skipped
   942→        for (var j = i; j < total; j++) {
   943→          actionable[j].importResult = 'skipped';
   944→        }
   945→        skipped += (total - i);
   946→        cancelled = true;
   947→        break;
   948→      }
   949→
   950→      final item = actionable[i];
   951→      final wasUpdate = item.status == BatchItemStatus.willUpdate;
   952→
   953→      onProgress?.call(BatchImportProgress(
   954→        current: i + 1,
   955→        total: total,
   956→        currentTitle: item.displayTitle,
   957→        successCreateCount: created,
   958→        successUpdateCount: updated,
   959→        failureCount: failed,
   960→      ));
   961→
   962→      try {
   963→        // addMovie() does its own duplicate re-check and counter sync.
   964→        // It returns the doc ID. We don't need it, but the call is required
   965→        // to trigger all the safety logic in FirestoreContentService.
   966→        //
   967→        // skipAdminCheck:true — admin was verified once at the start of
   968→        // runImport() (see top of this method). Skipping the per-item admin
   969→        // read saves N Firestore reads for an N-item import. See audit C2.
   970→        await _contentService.addMovie(item.data, skipAdminCheck: true);
   971→
   972→        if (wasUpdate) {
   973→          updated++;
   974→          item.importResult = 'success_update';
   975→        } else {
   976→          // addMovie() may have flipped this to an update if a race condition
   977→          // inserted a duplicate between classify() and now. We can detect
   978→          // this by checking if the returned data has createdAt set by us
   979→          // (no) vs the server (yes for new). For simplicity, trust the
   980→          // pre-classification — the discrepancy is at most one item.
   981→          created++;
   982→          item.importResult = 'success_create';
   983→        }
   984→      } catch (e) {
   985→        failed++;
   986→        item.importResult = 'failure';
   987→        item.importError = e.toString();
   988→        debugPrint('BatchImport item ${item.sourceIndex} failed: $e');
   989→        // Continue to next item — do not abort the whole batch.
   990→      }
   991→
   992→      onProgress?.call(BatchImportProgress(
   993→        current: i + 1,
   994→        total: total,
   995→        currentTitle: item.displayTitle,
   996→        successCreateCount: created,
   997→        successUpdateCount: updated,
   998→        failureCount: failed,
   999→        lastItem: item,
  1000→      ));
  1001→    }
  1002→
  1003→    final completedAt = DateTime.now();
  1004→    final result = BatchImportResult(
  1005→      total: items.length,
  1006→      created: created,
  1007→      updated: updated,
  1008→      failed: failed,
  1009→      skipped: skipped,
  1010→      items: items,
  1011→    );
  1012→
  1013→    // Record audit log entry. Failures here MUST NOT propagate — the import
  1014→    // itself has already happened and the user needs to see its result.
  1015→    if (auditContext != null) {
  1016→      try {
  1017→        await _recordAudit(
  1018→          context: auditContext,
  1019→          startedAt: startedAt,
  1020→          completedAt: completedAt,
  1021→          result: result,
  1022→          cancelled: cancelled,
  1023→        );
  1024→      } catch (e) {
  1025→        debugPrint('BatchImport audit log write failed (non-fatal): $e');
  1026→      }
  1027→    }
  1028→
  1029→    return result;
  1030→  }
  1031→
  1032→  /// Re-import only the items that failed in a previous [runImport] call.
  1033→  ///
  1034→  /// Pass the SAME [items] list that was returned in [BatchImportResult.items]
  1035→  /// — this method filters internally to just those whose `importResult ==
  1036→  /// 'failure'`. Items that previously succeeded are NOT touched (they're
  1037→  /// already in Firestore, and addMovie() would just no-op them anyway, but
  1038→  /// skipping them is faster and produces a cleaner audit record).
  1039→  ///
  1040→  /// Each failed item's `importResult` / `importError` are reset to null so
  1041→  /// the new attempt gets a fresh outcome. The original item `status`
  1042→  /// (willCreate / willUpdate) is preserved — addMovie() re-checks duplicates
  1043→  /// anyway, so a stale classification is harmless.
  1044→  ///
  1045→  /// If [auditContext] is provided, a NEW audit doc is written for the retry
  1046→  /// run with `isRetry: true`. This way the history list shows both the
  1047→  /// original import and the retry as separate entries — useful when
  1048→  /// investigating what fixed (or didn't fix) a failure.
  1049→  ///
  1050→  /// Returns a [BatchImportResult] that covers ONLY the retried items —
  1051→  /// `total` is the count of items retried (not the original batch size),
  1052→  /// and `items` is the filtered list with fresh outcomes. The caller can
  1053→  /// merge this back into its own state or display it standalone.
  1054→  Future<BatchImportResult> retryFailed(
  1055→    List<BatchImportItem> items, {
  1056→    void Function(BatchImportProgress)? onProgress,
  1057→    bool Function()? shouldStop,
  1058→    BatchImportAuditContext? auditContext,
  1059→  }) async {
  1060→    // Filter to only the failed items. We use a growable=false list for
  1061→    // consistency with runImport() and to avoid accidental mutation.
  1062→    final failedItems = items
  1063→        .where((i) => i.importResult == 'failure')
  1064→        .toList(growable: false);
  1065→
  1066→    if (failedItems.isEmpty) {
  1067→      // Nothing to retry — return an empty result. The caller should normally
  1068→      // not even show a Retry button in this case, but we handle it safely.
  1069→      return const BatchImportResult(
  1070→        total: 0,
  1071→        created: 0,
  1072→        updated: 0,
  1073→        failed: 0,
  1074→        skipped: 0,
  1075→        items: [],
  1076→      );
  1077→    }
  1078→
  1079→    // Reset failure state so we get fresh results from this attempt.
  1080→    // Status (willCreate/willUpdate) is preserved — addMovie() re-checks.
  1081→    for (final item in failedItems) {
  1082→      item.importResult = null;
  1083→      item.importError = null;
  1084→    }
  1085→
  1086→    // Re-use the existing import loop. Since failedItems all have a non-invalid
  1087→    // status (they made it past validation but failed at addMovie()), none
  1088→    // will be filtered out by runImport's `status != invalid` check.
  1089→    // runImport will record its own audit entry using the provided context.
  1090→    return runImport(
  1091→      failedItems,
  1092→      onProgress: onProgress,
  1093→      shouldStop: shouldStop,
  1094→      auditContext: auditContext,
  1095→    );
  1096→  }

  1099→  // PHASE 4 — AUDIT LOG

  1101→
  1102→  /// Firestore collection name where one document is written per import run.
  1103→  /// Path: `batch_imports/{importId}`.
  1104→  static const String auditCollectionName = 'batch_imports';
  1105→
  1106→  /// Write a single audit document to `batch_imports` describing the import
  1107→  /// that just finished. The document ID is auto-generated by Firestore.
  1108→  ///
  1109→  /// Schema (see [BatchImportAuditContext] + [BatchImportResult]):
  1110→  ///   {
  1111→  ///     "startedAt":        Timestamp,
  1112→  ///     "completedAt":      Timestamp,
  1113→  ///     "durationMs":       int,
  1114→  ///     "cancelled":        bool,
  1115→  ///     "adminUid":         String,
  1116→  ///     "adminEmail":       String?,
  1117→  ///     "sourceFileName":   String?,
  1118→  ///     "sourceFileSizeBytes": int?,
  1119→  ///     "appVersion":       String?,
  1120→  ///     "total":            int,
  1121→  ///     "created":          int,
  1122→  ///     "updated":          int,
  1123→  ///     "failed":           int,
  1124→  ///     "skipped":          int,
  1125→  ///     "failedItems":      [{sourceIndex,title,tmdbId,type,error}, ...],
  1126→  ///     "sampleCreated":    [String],   // up to 20 titles
  1127→  ///     "sampleUpdated":    [String]    // up to 20 titles
  1128→  ///   }
  1129→  Future<void> _recordAudit({
  1130→    required BatchImportAuditContext context,
  1131→    required DateTime startedAt,
  1132→    required DateTime completedAt,
  1133→    required BatchImportResult result,
  1134→    required bool cancelled,
  1135→  }) async {
  1136→    // Build the failed-items payload. Cap at 200 entries — anything bigger
  1137→    // would push the doc above Firestore's 1 MB limit and isn't useful to
  1138→    // review inline anyway (admin should re-import to fix).
  1139→    final failedItems = <Map<String, dynamic>>[];
  1140→    final failedFull =
  1141→        result.items.where((i) => i.importResult == 'failure').toList();
  1142→    for (final item in failedFull.take(200)) {
  1143→      failedItems.add({
  1144→        'sourceIndex': item.sourceIndex,
  1145→        'title': item.displayTitle,
  1146→        'tmdbId': item.tmdbId,
  1147→        'type': item.displayType,
  1148→        'error': item.importError,
  1149→      });
  1150→    }
  1151→
  1152→    // Capture up to 20 sample titles for each outcome so the history detail
  1153→    // page can show "what got created" without re-fetching all movies.
  1154→    final sampleCreated = result.items
  1155→        .where((i) => i.importResult == 'success_create')
  1156→        .take(20)
  1157→        .map((i) => i.displayTitle)
  1158→        .toList();
  1159→    final sampleUpdated = result.items
  1160→        .where((i) => i.importResult == 'success_update')
  1161→        .take(20)
  1162→        .map((i) => i.displayTitle)
  1163→        .toList();
  1164→
  1165→    final payload = <String, dynamic>{
  1166→      ...context.toPartialFirestoreMap(),
  1167→      'startedAt': startedAt.toUtc(),
  1168→      'completedAt': completedAt.toUtc(),
  1169→      'durationMs': completedAt.difference(startedAt).inMilliseconds,
  1170→      'cancelled': cancelled,
  1171→      'total': result.total,
  1172→      'created': result.created,
  1173→      'updated': result.updated,
  1174→      'failed': result.failed,
  1175→      'skipped': result.skipped,
  1176→      'failedItems': failedItems,
  1177→      'sampleCreated': sampleCreated,
  1178→      'sampleUpdated': sampleUpdated,
  1179→    };
  1180→
  1181→    await _firestore.collection(auditCollectionName).add(payload);
  1182→  }

  1243→  /// Deep-cast a JSON map (which may contain dynamic values from jsonDecode)
  1244→  /// into a `Map<String, dynamic>` with proper types. Lists are recursively
  1245→  /// cast so `List<dynamic>` becomes `List<Map<String, dynamic>>` where
  1246→  /// appropriate — this is what FirestoreContentService.addMovie() expects.
  1247→  Map<String, dynamic> _deepCast(Map<dynamic, dynamic> src) {
  1248→    final out = <String, dynamic>{};
  1249→    src.forEach((key, value) {
  1250→      out[key.toString()] = _deepCastValue(value);
  1251→    });
  1252→    return out;
  1253→  }
  1254→
  1255→  dynamic _deepCastValue(dynamic value) {
  1256→    if (value is Map) {
  1257→      return _deepCast(value);
  1258→    }
  1259→    if (value is List) {
  1260→      return value.map(_deepCastValue).toList();
  1261→    }
  1262→    return value;
  1263→  }

  1265→  /// Validate that the item has the minimal fields required by addMovie().
  1266→  /// Returns an error string if invalid, null if OK.
  1267→  ///
  1268→  /// Validation rules:
  1269→  ///   - `title` must be a non-empty string.
  1270→  ///   - `type` (if present) must be 'movie' or 'series'.
  1271→  ///   - `tmdbId` (if present) must be an int or a numeric string.
  1272→  ///   - `categories` and `tags` (if present) must be Lists of strings.
  1273→  String? _validate(Map<String, dynamic> data) {
  1274→    final title = data['title'];
  1275→    if (title is! String || title.trim().isEmpty) {
  1276→      return 'Missing or empty "title" field.';
  1277→    }
  1278→
  1279→    final type = data['type'];
  1280→    if (type != null && type is! String) {
  1281→      return '"type" must be a string ("movie" or "series").';
  1282→    }
  1283→    if (type is String && type != 'movie' && type != 'series') {
  1284→      return '"type" must be "movie" or "series" (got "$type").';
  1285→    }
  1286→
  1287→    final tmdbId = data['tmdbId'];
  1288→    if (tmdbId != null) {
  1289→      if (tmdbId is! int && tmdbId is! num) {
  1290→        // Allow numeric strings; non-numeric strings are an error.
  1291→        if (tmdbId is String) {
  1292→          if (int.tryParse(tmdbId) == null) {
  1293→            return '"tmdbId" must be an integer or numeric string '
  1294→                '(got "$tmdbId").';
  1295→          }
  1296→          // Promote to int — addMovie() does numeric comparison.
  1297→          data['tmdbId'] = int.parse(tmdbId);
  1298→        } else {
  1299→          return '"tmdbId" must be an integer (got ${tmdbId.runtimeType}).';
  1300→        }
  1301→      }
  1302→    }
  1303→
  1304→    final categories = data['categories'];
  1305→    if (categories != null && categories is! List) {
  1306→      return '"categories" must be an array of strings.';
  1307→    }
  1308→
  1309→    final tags = data['tags'];
  1310→    if (tags != null && tags is! List) {
  1311→      return '"tags" must be an array of strings.';
  1312→    }
  1313→
  1314→    return null;
  1315→  }

  1317→  /// Slug generator — mirrors FirestoreContentService._generateSlug.
  1318→  /// Used during classify() to detect slug duplicates for items without a
  1319→  /// pre-set slug. Kept private; addMovie() will auto-generate the real slug.
  1320→  String _generateSlug(String title) {
  1321→    return title
  1322→        .toLowerCase()
  1323→        .replaceAll(RegExp(r'[^\w\s-]'), '')
  1324→        .replaceAll(RegExp(r'\s+'), '-')
  1325→        .replaceAll(RegExp(r'-+'), '-')
  1326→        .trim();
  1327→  }
