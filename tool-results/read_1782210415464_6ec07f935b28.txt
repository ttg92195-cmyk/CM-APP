   447→   447→
   448→   448→class _FilterResultPageState extends State<FilterResultPage> {
   449→   449→  final FirestoreContentService _contentService = FirestoreContentService();
   450→   450→
   451→   451→  List<Movie> _movies = [];
   452→   452→  List<Movie> _filteredMovies = [];
   453→   453→  DocumentSnapshot? _lastDoc;
   454→   454→  bool _hasMore = true;
   455→   455→  bool _isLoading = true;
   456→   456→  bool _isLoadingMore = false;
   457→   457→  // Track seen IDs to prevent duplicates
   458→   458→  final Set<String> _seenIds = {};
   459→   459→
   460→   460→  @override
   461→   461→  void initState() {
   462→   462→    super.initState();
   463→   463→    _loadMovies();
   464→   464→  }
   465→   465→
   466→   466→  Future<void> _loadMovies() async {
   467→   467→    setState(() {
   468→   468→      _isLoading = true;
   469→   469→      _seenIds.clear();
   470→   470→    });
   471→   471→    try {
   472→   472→      Map<String, dynamic> result;
   473→   473→      if (widget.genreName != null) {
   474→   474→        result = await _contentService.getMoviesByGenre(
   475→   475→          widget.genreName!, limit: 20, typeFilter: widget.typeFilter, // PAGINATION: 20/page (Task 38 Req 5)
   476→   476→        );
   477→   477→      } else if (widget.tagName != null) {
   478→   478→        result = await _contentService.getMoviesByTag(
   479→   479→          widget.tagName!, limit: 20, typeFilter: widget.typeFilter, // PAGINATION: 20/page (Task 38 Req 5)
   480→   480→        );
   481→   481→      } else if (widget.collectionName != null) {
   482→   482→        result = await _contentService.getMoviesByCollection(
   483→   483→          widget.collectionName!, limit: 20, typeFilter: widget.typeFilter, // PAGINATION: 20/page (Task 38 Req 5)
   484→   484→        );
   485→   485→      } else {
   486→   486→        result = await _contentService.getMovies(limit: 20); // PAGINATION: 20/page
   487→   487→      }
   488→   488→
   489→   489→      if (mounted) {
   490→   490→        final allMovies = result['movies'] as List<Movie>;
   491→   491→
   492→   492→        // Track IDs to prevent future duplicates
   493→   493→        for (final m in allMovies) {
   494→   494→          _seenIds.add(m.id);
   495→   495→        }
   496→   496→
   497→   497→        // Apply type filter if specified (Movies or Series sub-tab)
   498→   498→        List<Movie> filtered;
   499→   499→        if (widget.typeFilter != null) {
   500→   500→          filtered = allMovies
   501→   501→              .where((m) => m.type == widget.typeFilter)
   502→   502→              .toList();
   503→   503→        } else {
   504→   504→          filtered = allMovies;
   505→   505→        }
   506→   506→
   507→   507→        // NOTE: previously had an artificial 600ms skeleton floor here.
   508→   508→        // Removed in Task 36 #2 — see movies_page.dart for the rationale.
   509→   509→        setState(() {
   510→   510→          _movies = allMovies;
   511→   511→          _filteredMovies = filtered;
   512→   512→          _hasMore = result['hasMore'] as bool;
   513→   513→          _lastDoc = result['lastDoc'] as DocumentSnapshot?;
   514→   514→          _isLoading = false;
   515→   515→        });
   516→   516→
   517→   517→        // Task 38 Req 5 — post-frame auto-load safety net.
   518→   518→        // If the first page returned fewer items than would fill the
   519→   519→        // viewport (e.g., on a large tablet showing 30+ items per screen,
   520→   520→        // or a genre with exactly 20 docs that doesn't fill the screen),
   521→   521→        // the scroll listener never fires — and infinite scroll silently
   522→   522→        // dies. Schedule a `_loadMore()` on the next frame; `_loadMore`
   523→   523→        // will no-op if `_hasMore` is already false (e.g., the genre only
   524→   524→        // had 20 docs total).
   525→   525→        if (_hasMore && _filteredMovies.length < 30) {
   526→   526→          WidgetsBinding.instance.addPostFrameCallback((_) {
   527→   527→            if (mounted && _hasMore && !_isLoadingMore) {
   528→   528→              _loadMore();
   529→   529→            }
   530→   530→          });
   531→   531→        }
   532→   532→      }
   533→   533→    } catch (e) {
   534→   534→      debugPrint('FilterResultPage load error: $e');
   535→   535→      if (mounted) setState(() => _isLoading = false);
   536→   536→    }
   537→   537→  }
   538→   538→
   539→   539→  Future<void> _loadMore() async {
   540→   540→    if (_isLoadingMore || !_hasMore) return;
   541→   541→    setState(() => _isLoadingMore = true);
   542→   542→    try {
   543→   543→      Map<String, dynamic> result;
   544→   544→      if (widget.genreName != null) {
   545→   545→        result = await _contentService.getMoviesByGenre(
   546→   546→          widget.genreName!, limit: 20, startAfter: _lastDoc,
   547→   547→          typeFilter: widget.typeFilter, // Task 38 Req 5
   548→   548→        );
   549→   549→      } else if (widget.tagName != null) {
   550→   550→        result = await _contentService.getMoviesByTag(
   551→   551→          widget.tagName!, limit: 20, startAfter: _lastDoc,
   552→   552→          typeFilter: widget.typeFilter, // Task 38 Req 5
   553→   553→        );
   554→   554→      } else if (widget.collectionName != null) {
   555→   555→        result = await _contentService.getMoviesByCollection(
   556→   556→          widget.collectionName!, limit: 20, startAfter: _lastDoc,
   557→   557→          typeFilter: widget.typeFilter, // Task 38 Req 5
   558→   558→        );
   559→   559→      } else {
   560→   560→        result = await _contentService.getMovies(limit: 20, startAfter: _lastDoc);
   561→   561→      }
   562→   562→
   563→   563→      if (mounted) {
   564→   564→        final newMovies = result['movies'] as List<Movie>;
   565→   565→
   566→   566→        // Deduplicate by ID to prevent duplicates
   567→   567→        final dedupedMovies = <Movie>[];
   568→   568→        for (final m in newMovies) {
   569→   569→          if (!_seenIds.contains(m.id)) {
   570→   570→            _seenIds.add(m.id);
   571→   571→            dedupedMovies.add(m);
   572→   572→          }
   573→   573→        }
   574→   574→
   575→   575→        List<Movie> filtered;
   576→   576→        if (widget.typeFilter != null) {
   577→   577→          filtered = dedupedMovies
   578→   578→              .where((m) => m.type == widget.typeFilter)
   579→   579→              .toList();
   580→   580→        } else {
   581→   581→          filtered = dedupedMovies;
   582→   582→        }
   583→   583→
   584→   584→        setState(() {
   585→   585→          _movies.addAll(dedupedMovies);
   586→   586→          _filteredMovies.addAll(filtered);
   587→   587→          _hasMore = result['hasMore'] as bool && newMovies.isNotEmpty;
   588→   588→          _lastDoc = result['lastDoc'] as DocumentSnapshot?;
   589→   589→          _isLoadingMore = false;
   590→   590→        });
   591→   591→
   592→   592→        // Task 38 Req 5 — chained auto-load safety net (mirror of the one
   593→   593→        // in _loadMovies). If after a `_loadMore` we still have fewer
   594→   594→        // than 30 visible items AND `_hasMore` is still true, the grid
   595→   595→        // likely still doesn't fill the viewport — schedule another
   596→   596→        // `_loadMore` on the next frame. Caps infinite recursion because
   597→   597→        // `_loadMore` no-ops once `_hasMore` flips false (Firestore
   598→   598→        // returned fewer than `limit` docs).
   599→   599→        if (_hasMore && _filteredMovies.length < 30) {
   600→   600→          WidgetsBinding.instance.addPostFrameCallback((_) {
   601→   601→            if (mounted && _hasMore && !_isLoadingMore) {
   602→   602→              _loadMore();
   603→   603→            }
   604→   604→          });
   605→   605→        }
   606→   606→      }
   607→   607→    } catch (e) {
   608→   608→      if (mounted) setState(() => _isLoadingMore = false);
   609→   609→    }
   610→   610→  }
